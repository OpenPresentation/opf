// Builds spec/reference/layout-migration.json from the 0.18 records (spec/catalogs/layouts), the targets of the design
// doc's mapping table and the 28 templates, and checks every row against the table (design section 9).
import fs from 'node:fs';
const repo = 'C:/opf-work/rr79/opf/';
const doc = fs.readFileSync('C:/opf-work/rr79/design.md', 'utf8');
const templates = JSON.parse(fs.readFileSync(repo + 'packages/javascript/test/fixtures/layout-templates-0.19.json', 'utf8')).layouts;
const rows = new Map();
for (const line of doc.split('\n')) {
  const m = /^\| `([a-z0-9-]+)` \| `([a-z-]+)` \| ([^|]*)\|([^|]*)\|$/.exec(line);
  if (!m) continue;
  const settings = {};
  for (const part of m[3].split(',').map(text => text.trim()).filter(Boolean)) {
    const s = /^`([a-zA-Z]+): ([a-z]+)`$/.exec(part);
    if (!s) throw new Error(`${m[1]}: ${part}`);
    settings[s[1]] = s[2] === 'true' ? true : s[2] === 'false' ? false : s[2];
  }
  rows.set(m[1], { layout: m[2], settings, content: m[4].trim() });
}
const ENGINE = { titleAlignment: 'left', contentAlignment: 'left', contentBox: false, listBullet: 'character', imageFit: 'cover' };
const KEYS = ['titleAlignment', 'contentAlignment', 'contentBox', 'contentDirection', 'listBullet', 'imageFit'];
const dir = repo + 'spec/catalogs/layouts/';
const out = {};
let problems = 0;
const ids = fs.readdirSync(dir).filter(file => file.endsWith('.json') && file !== 'index.json').map(file => file.slice(0, -5)).sort();
for (const id of ids) {
  const old = JSON.parse(fs.readFileSync(dir + id + '.json', 'utf8'));
  const row = rows.get(id);
  if (!row) { console.log('no row', id); problems++; continue; }
  const target = row.layout;
  const own = target === 'auto' ? {} : templates[target].design ?? {};
  const design = {};
  for (const key of KEYS) {
    const value = old.design?.[key];
    if (value === undefined) continue;
    if (key === 'contentDirection') { if (value === 'vertical') design[key] = value; continue; }
    const theirs = own[key] ?? ENGINE[key];
    if (value !== theirs) design[key] = value;
  }
  if (old.design?.chartPrimary !== undefined) console.log('chartPrimary on', id, old.design.chartPrimary);
  const docSettings = row.settings;
  const same = JSON.stringify(Object.keys(design).sort().map(key => [key, design[key]])) === JSON.stringify(Object.keys(docSettings).sort().map(key => [key, docSettings[key]]));
  if (!same) { console.log('MISMATCH', id, JSON.stringify(design), JSON.stringify(docSettings)); problems++; }
  const entry = { layout: target };
  if (Object.keys(design).length) entry.design = Object.fromEntries(KEYS.filter(key => key in design).map(key => [key, design[key]]));
  const body = (old.placeholders ?? []).filter(p => !['title', 'subtitle', 'tag'].includes(p.type));
  const strip = (entries) => entries.map(p => p.type === 'group' ? { type: 'group', ...(p.composition && Object.keys(p.composition).length ? { composition: p.composition } : {}), placeholders: strip(p.placeholders) } : { type: p.type });
  if (row.content === 'groups') entry.content = { groups: strip(body) };
  else if (row.content.startsWith('placement')) {
    const placed = body.filter(p => p.placement);
    if (placed.length !== 1) throw new Error(id);
    const index = body.filter(p => p.type === 'image').indexOf(placed[0]);
    entry.content = { placement: { image: index, ...placed[0].placement } };
    if (`placement \`${placed[0].placement.edge}\`` !== row.content) { console.log('PLACEMENT MISMATCH', id, row.content); problems++; }
  } else if (row.content === 'drop `layout`') entry.content = { dropLayout: true };
  else if (row.content) { console.log('content?', id, row.content); problems++; }
  out[id] = entry;
}
for (const id of rows.keys()) if (!ids.includes(id)) { console.log('row without record', id); problems++; }
const json = {
  $comment: 'OPF 0.19 (design section 9): every layout id of the 0.18 default catalog that 0.19 removes, with the built-in layout and the design settings that reproduce its variant, and the content rewrites of the records with placeholder groups (groups: the record\'s body placeholders, which wrap the slide\'s blocks into content groups), an image placement (placement: the n-th top-level image takes this band) or none at all (dropLayout). The design doc\'s mapping table is generated from this file. opf/layout-removed and migrate() read it.',
  version: 1,
  layouts: out,
};
const counts = {};
for (const entry of Object.values(out)) for (const [key, value] of Object.entries(entry.design ?? {})) counts[`${key}: ${value}`] = (counts[`${key}: ${value}`] ?? 0) + 1;
console.log(Object.keys(out).length, 'rows', Object.values(out).filter(e => e.design).length, 'with settings', JSON.stringify(counts));
console.log('groups', Object.values(out).filter(e => e.content?.groups).length, 'placement', Object.values(out).filter(e => e.content?.placement).length, 'drop', Object.values(out).filter(e => e.content?.dropLayout).length);
if (problems) { console.log(problems, 'problems'); process.exit(1); }
fs.writeFileSync(repo + 'spec/reference/layout-migration.json', JSON.stringify(json, null, 2) + '\n');
