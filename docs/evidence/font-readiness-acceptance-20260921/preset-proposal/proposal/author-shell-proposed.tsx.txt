let accountGallerySourcesCache: AuthorGallerySource[] | null = null;
let accountGallerySourcesRequest: Promise<AuthorGallerySource[]> | null = null;

function fetchAccountGallerySourcesOnce(): Promise<AuthorGallerySource[]> {
  if (accountGallerySourcesCache) return Promise.resolve(accountGallerySourcesCache);
  accountGallerySourcesRequest ??= fetch("/api/v1/author/gallery-sources")
    .then(async (response) => {
      if (!response.ok) throw new Error(`Gallery sources returned HTTP ${response.status}`);
      return response.json();
    })
    .then((payload) => {
      const sources =
        isRecord(payload) && Array.isArray(payload.sources)
          ? (payload.sources as AuthorGallerySource[])
          : [];
      accountGallerySourcesCache = sources;
      return sources;
    })
    .catch((error: unknown) => {
      accountGallerySourcesRequest = null;
      throw error;
    });
  return accountGallerySourcesRequest;
}

function authorMessagesSyncKey(messages: PublicAuthorMessage[], runs: PublicAuthorRun[]): string {
  return [
    ...messages.map((message) =>
      [
        "message",
        message.id,
        message.role,
        message.operation,
        message.runId ?? "",
        message.createdAt,
        message.content,
      ].join(":")
    ),
    ...runs.map((run) =>
      [
        "run",
        run.id,
        run.status,
        run.updatedAt,
        run.toolMeta?.toolCalls?.length ?? 0,
        run.toolMeta?.proposal?.createdAt ?? "",
        typeof run.modelMeta?.reason === "string" ? run.modelMeta.reason : "",
      ].join(":")
    ),
  ].join("\n");
}

function useWideAuthorLayout(): boolean {
  const [wide, setWide] = useState(false);
  useEffect(() => {
    if (typeof window === "undefined") return;
    const query = window.matchMedia("(min-width: 1024px)");
    const apply = () => setWide(query.matches);
    apply();
    query.addEventListener("change", apply);
    return () => query.removeEventListener("change", apply);
  }, []);
  return wide;
}

function buildPreviewState(parse: ReturnType<typeof parseDocument>): PreviewState {
  if (!parse.document) {
    if (parse.error) {
      return {
        status: "schema_error",
        errors: [{ path: "$", message: parse.error }],
      };
    }
    return { status: "idle" };
  }
  const v = validateOPF(parse.document);
  if (!v.valid) {
    return {
      status: "schema_error",
      errors: v.errors.map((item) => ({ path: item.path, message: item.message })),
    };
  }
  return { status: "ok", doc: parse.document };
}

function AuthorOverflowMenu({
  documentId,
  documentActions = [],
  onCopyId,
  onShareLink,
}: {
  documentId?: string | null;
  documentActions?: ReadonlyArray<OverflowMenuItem>;
  onCopyId?: () => void;
  onShareLink?: () => void;
}) {
  const documentItems: OverflowMenuItem[] = [];

  if (documentId && onCopyId) {
    documentItems.push({
      kind: "action",
      label: "Copy document id",
      shortcut: "⌘I",
      onSelect: onCopyId,
    });
  }
  if (onShareLink) {
    documentItems.push({
      kind: "action",
      label: "Share workspace link",
      shortcut: "⌘⇧S",
      onSelect: onShareLink,
    });
  }
  documentItems.push(...documentActions);

  const sections: OverflowMenuSection[] = [];
  if (documentItems.length > 0) {
    sections.push({ title: "Document", items: documentItems });
  }

  sections.push({
    title: "Reference",
    items: [
      { kind: "link", label: "Docs", href: "/docs" },
      { kind: "link", label: "API reference", href: "/docs/api" },
      { kind: "link", label: "OPF schema · v1", href: "/docs/opf" },
      {
        kind: "link",
        label: "pptx.gallery",
        href: "https://pptx.gallery",
        external: true,
      },
    ],
  });

  return <OverflowMenu sections={sections} />;
}

type AuthorSource = { text: string; format: Format };
type AuthorDraft = AuthorSource & { document: OPFDocument };
type PresetGroup = {
  editor: EditorSession;
  original: AuthorSource;
  expected: AuthorSource;
  undoDepth: number;
  redoDepth: number;
};
type PresetTransaction = {
  token: object;
  editor: EditorSession;
  before: OPFDocument;
  after: OPFDocument;
  expected: AuthorSource;
  result: AuthorSource;
  undoDepth: number;
  override?: CanvasSourceOverride;
  accepted: boolean;
};

function matchesPresetGroup(group: PresetGroup, source: AuthorSource, editor: EditorSession | null): boolean {
  if (!editor || group.editor !== editor || group.expected.text !== source.text || group.expected.format !== source.format) return false;
  const snapshot = editor.snapshot();
  return snapshot.undoDepth === group.undoDepth && snapshot.redoDepth === group.redoDepth;
}

  // Uses the backend endpoint, which prefers the AI path and falls back
  // to deterministic if AI is disabled or fails.
  //
  // `authorIntent` is intentionally NOT in the deps array: it changes on
  // every keystroke, which would cancel and re-skip the fetch (the run-id

  // is read through a ref so the in-flight fetch sees the latest value
  // but typing mid-fetch doesn't void the request.
  //
  // Out-of-order safety: we don't use a `cancelled` flag because the
  // cleanup would otherwise strand `followUpLoading=true` whenever the
  // effect re-runs and bails at the dedup check. Instead the async block
  // gates state writes on `followUpRunRef.current === runIdAtFetchTime`,
  // so a late-arriving response for a superseded run is dropped while a
  // mid-flight fetch for the *same* run still completes correctly.

    const applied = runs.find((run) => run.status === "applied");
    if (!applied) {
      setFollowUpChips([]);
      setFollowUpLoading(false);
      followUpRunRef.current = null;
      return;
    }
    if (!parseDraftState.document) {
      // OPF is mid-edit (invalid JSON). Don't clear the dedup ref or the
      // chips — typing past a parse error and back would otherwise reset
      // the guard and re-fire the fetch for the same applied run.
      return;
    }
    if (followUpRunRef.current === applied.id) {
      // Already fetched (or fetching) chips for this run. Don't touch
      // `followUpLoading` — the original fetch's `finally` clears it via
      // the run-id gate. Resetting it here while the fetch is still in
      // flight would create a brief gap where neither the loading
      // indicator nor the chips are visible.
      return;
    }
    followUpRunRef.current = applied.id;
    const runIdAtFetchTime = applied.id;
    const opfForChips = parseDraftState.document;
    setFollowUpLoading(true);

    void (async () => {
      try {
        const response = await fetch("/api/v1/author/follow-ups", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({
            intent: authorIntentRef.current,
            opf: opfForChips,
            lastAssistantText: applied.assistantText,
            lastOperation: applied.operation,
          }),
        });
        if (!response.ok) throw new Error(`follow-ups ${response.status}`);
        const payload = (await response.json()) as { chips?: FollowUpChip[] };
        if (followUpRunRef.current === runIdAtFetchTime && Array.isArray(payload.chips)) {
          setFollowUpChips(payload.chips);
        }
      } catch {
        if (followUpRunRef.current === runIdAtFetchTime) {
          setFollowUpChips(
            deterministicFollowUps({
              intent: authorIntentRef.current,
              opf: opfForChips,
              lastAssistantText: applied.assistantText,
              lastOperation: applied.operation as "create" | "refine" | "ask",
            })
          );
        }
      } finally {
        if (followUpRunRef.current === runIdAtFetchTime) {
          setFollowUpLoading(false);
        }
      }
    })();

    // (see the comment block above for why). Because the ref itself is

  }, [runs, parseDraftState.document]);

  const commitPresetDocument = useCallback((draft: AuthorDraft, nextDocument: OPFDocument, rawResult?: AuthorSource) => {
    let pending: PresetTransaction | null = null;
    try {
      const editor = canvasSession.editor;
      if (isSending || !editor || readySessionRef.current !== editor || presetTransactionRef.current
        || sourceRef.current.text !== draft.text || sourceRef.current.format !== draft.format
        || !sameJsonValue(editor.document, draft.document)) {
        throw new Error("The draft changed before this preset edit could be applied. Reopen the current draft and try again.");
      }
      for (const document of [draft.document, nextDocument]) {
        const checked = validateOPF(document);
        if (!checked.valid) throw new Error(checked.errors[0]?.message ?? "Preset edit requires valid OPF.");
      }
      if (sameJsonValue(draft.document, nextDocument)
        && (!rawResult || (rawResult.text === draft.text && rawResult.format === draft.format))) {
        return { changed: false, editor, source: { text: draft.text, format: draft.format }, snapshot: editor.snapshot() };
      }
      let result: AuthorSource;
      if (rawResult) {
        if (rawResult.format !== draft.format) throw new Error("The source format changed after these presets were applied.");
        result = { ...rawResult };
        if (draft.format === "json") {
          updateJsonSource(draft.text, draft.document);
          updateJsonSource(result.text, nextDocument);
        }
      } else if (draft.format === "json") {
        result = { text: updateJsonSource(draft.text, nextDocument), format: draft.format };
      } else {
        const converted = convert(toJsonText(nextDocument), "json", draft.format);
        if (!converted.ok) throw new Error(converted.error);
        result = { text: converted.text, format: draft.format };
      }
      const parsed = parseText(result.text, result.format);
      if (!parsed.ok || !sameJsonValue(parsed.doc, nextDocument)) {
        throw new Error("This preset edit cannot be represented in the current source format. Your source was kept.");
      }
      const expected = { text: draft.text, format: draft.format };
      pending = {
        token: {}, editor, before: draft.document, after: nextDocument, expected, result,
        undoDepth: editor.snapshot().undoDepth + 1,
        override: rawResult ? { expected, result } : undefined,
        accepted: false,
      };
      const rejectedBeforeCommit = rejectedCanvasChangeRef.current;
      presetTransactionRef.current = pending;
      editor.applyPatch([{ op: "replace", path: "", value: nextDocument }], {
        source: "author-preset", authorPresetTransaction: pending.token, rejectInvalid: true,
      });
      if (!pending.accepted || rejectedCanvasChangeRef.current !== rejectedBeforeCommit
        || readySessionRef.current !== editor
        || sourceRef.current.text !== result.text || sourceRef.current.format !== result.format
        || !sameJsonValue(editor.document, nextDocument)) {
        throw new Error("The preset edit was not accepted by the current draft. Your source was kept.");
      }
      setSurfaceError(null);
      return { changed: true, editor, source: result, snapshot: editor.snapshot() };
    } catch (error) {
      invalidatePresetGroup();
      setSurfaceError(error instanceof Error ? error.message : "Could not apply the preset edit.");
      return null;
    } finally {
      if (presetTransactionRef.current === pending) presetTransactionRef.current = null;
    }
  }, [canvasSession.editor, invalidatePresetGroup, isSending]);

      if (isSending) return;
      // An inline draft is committed before taking the group's initial source.
      const current = readCurrentDraft();
      if (!current?.document) return;

      const group = previousGroup && matchesPresetGroup(previousGroup, draft, canvasSession.editor) ? previousGroup : null;
      if (previousGroup && !group) invalidatePresetGroup();

        setSurfaceError(`Could not apply ${suggestion.label} — field is not deck-wide.`);
        return;

      if (!committed?.changed) return;

        editor: committed.editor,
        original: group?.original ?? { text: draft.text, format: draft.format },
        expected: committed.source,
        undoDepth: committed.snapshot.undoDepth,
        redoDepth: committed.snapshot.redoDepth,

        ...toasts,

      jumpToPath(applied.fieldPath);
    },
    [canvasSession.editor, commitPresetDocument, invalidatePresetGroup, isSending, jumpToPath, readCurrentDraft]

      invalidatePresetGroup();
      setSurfaceError("The original preset source is no longer valid. Your current draft was kept.");
      return;
    }
    const committed = commitPresetDocument({ ...current, document: current.document }, original.document, group.original);
    if (committed) invalidatePresetGroup();
  }, [canvasSession.editor, commitPresetDocument, invalidatePresetGroup, isSending, readCurrentDraft]);

function EditableFileChip({
  disabled,
  filename,
  onCommit,
}: {
  disabled: boolean;
  filename: string;
  onCommit: (value: string) => boolean;
}) {
  const [editing, setEditing] = useState(false);
  const [draft, setDraft] = useState(filename);
  const inputRef = useRef<HTMLInputElement | null>(null);
  const skipBlurCommitRef = useRef(false);

  useEffect(() => {
    if (!editing) return;
    inputRef.current?.focus();
    inputRef.current?.select();
  }, [editing]);

  const commit = useCallback(() => {
    const accepted = onCommit(draft);
    if (accepted) setEditing(false);
    return accepted;
  }, [draft, onCommit]);

  if (editing) {
    return (
      <span
        className="inline-flex h-6 max-w-[18rem] min-w-[12rem] shrink items-center gap-1.5 rounded-md border border-border bg-background px-2 font-mono text-[11px] text-muted-foreground focus-within:ring-2 focus-within:ring-ring"
        title="Edit filename"
      >
        <FileCode className="size-3 shrink-0" aria-hidden />
        <input
          ref={inputRef}
          aria-label="Filename"
          value={draft}
          onChange={(event) => setDraft(event.target.value)}
          onBlur={() => {
            if (skipBlurCommitRef.current) {
              skipBlurCommitRef.current = false;
              return;
            }
            commit();
          }}
          onKeyDown={(event) => {
            if (event.key === "Enter") {
              event.preventDefault();
              commit();
            }
            if (event.key === "Escape") {
              event.preventDefault();
              skipBlurCommitRef.current = true;
              setDraft(filename);
              setEditing(false);
            }
          }}
          className="min-w-0 flex-1 bg-transparent text-foreground outline-none"
        />
      </span>
    );
  }

  return (
    <button
      type="button"
      disabled={disabled}
      onClick={() => {
        if (!disabled) {
          setDraft(filename);
          setEditing(true);
        }
      }}
      className={cn(
        "inline-flex h-8 max-w-[18rem] min-w-0 shrink items-center gap-1.5 rounded-full border border-border bg-background px-3 font-mono text-xs text-muted-foreground transition-colors hover:bg-muted/40 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring disabled:cursor-default disabled:hover:bg-background",
        "h-6 rounded-md px-2 text-[11px]"
      )}
      title={disabled ? filename : "Edit filename"}
    >
      <FileCode className="size-3 shrink-0" aria-hidden />{" "}
      <span className="min-w-0 truncate text-foreground">{filename}</span>
    </button>
  );
}

function toastToPick(toast: PresetAppliedToastData): AutoAppliedPick {
  const value = renderPresetValue(toast.after) ?? toast.label;
  return {
    kind: fieldPathToKind(toast.fieldPath),
    value,
    reason: toast.label,
    hunkId: toast.id,
  };
}

function fieldPathToKind(path: string): string {
  // `$.design.theme` → "theme", `$.audience` → "audience".
  const parts = path.replace(/^\$\.?/, "").split(".");
  return parts[parts.length - 1] ?? path;
}

function renderPresetValue(value: unknown): string | null {
  if (value === undefined || value === null) return null;
  if (typeof value === "string") return value;
  if (typeof value === "number" || typeof value === "boolean") return String(value);
  if (value && typeof value === "object") {
    const obj = value as Record<string, unknown>;
    const candidate = obj.label ?? obj.name ?? obj.id ?? obj.slug ?? obj.scheme ?? obj.template;
    if (typeof candidate === "string") return candidate;
  }
  return null;
}

function dedupeTranscriptMessages(messages: AuthorUIMessage[]) {
  const seen = new Set<string>();
  const deduped: AuthorUIMessage[] = [];
  for (const message of messages) {
    const runId = message.metadata?.runId;
    const key = runId
      ? `${message.role}:${runId}:${textFromUIMessage(message)}`
      : `id:${message.id}`;
    if (seen.has(key)) continue;
    seen.add(key);
    deduped.push(message);
  }
  return deduped;
}

  onGhostRejectAll: () => void;
  onGhostAcceptHunk: (hunkId: string) => void;
  onGhostRejectHunk: (hunkId: string) => void;
  onGhostJump: (hunk: GhostHunk) => void;
  onOpenDiff: () => void;
  onToolApprovalResponse: (response: {
    id: string;
    approved: boolean;
    reason?: string;
  }) => void | PromiseLike<void>;
  presetToasts: ReadonlyArray<PresetAppliedToastData>;
  onPresetUndoAll: () => void;
  followUpChips: ReadonlyArray<FollowUpChip>;
  followUpLoading: boolean;
  onFollowUpPick: (chip: FollowUpChip) => void;
}) {
  const runById = useMemo(() => new Map(runs.map((run) => [run.id, run])), [runs]);
  const resolvedReviewMessageIds = useMemo(
    () =>
      new Set(
        runs
          .filter((run) => run.status === "applied" || run.status === "rejected")
          .map((run) => run.messageId)
      ),
    [runs]
  );
  const derivedMessages = useMemo(
    () =>
      messages.map((message) =>
        authorMessageToUIMessage(
          message,
          message.runId ? runById.get(message.runId) : undefined
        )
      ),
    [messages, runById]
  );
  const transcriptMessages = useMemo(
    () => dedupeTranscriptMessages(uiMessages?.length ? uiMessages : derivedMessages),
    [derivedMessages, uiMessages]
  );
  const messageUserMessageIds = useMemo(() => {
    const ids = new Map<string, string>();
    let currentUserMessageId: string | null = null;
    for (const message of transcriptMessages) {
      if (message.role === "user") currentUserMessageId = message.id;
      const runMessageId = message.run?.messageId;
      if (runMessageId) ids.set(message.id, runMessageId);
      else if (currentUserMessageId) ids.set(message.id, currentUserMessageId);
    }
    return ids;
  }, [transcriptMessages]);
  const latestActionableApprovalMessageId = useMemo(() => {
    let latest: string | null = null;
    for (const message of transcriptMessages) {
      if (
        message.parts.some(
          (part) => part.type === "tool-propose_edits" && part.state === "approval-requested"
        )
      ) {
        latest = message.id;
      }
      const text = textFromUIMessage(message);
      if (
        text.startsWith("Approved agent proposal:") ||
        text.startsWith("Rejected agent proposal:") ||
        message.run?.status === "applied" ||
        message.run?.status === "rejected"
      ) {
        latest = null;
      }
    }
    return latest;
  }, [transcriptMessages]);

  return (
    <Conversation className="min-h-0 flex-1">
      <ConversationContent className="gap-3 px-3 py-3">
        {transcriptMessages.length === 0 ? (
          <ConversationEmptyState
            className="min-h-44 justify-start pt-10 text-left"
            title="Agent transcript"
            description="Plans, tool calls, diffs, and checkpoints will appear here as Author works."
          />
        ) : null}
        {transcriptMessages.map((message) => (
          <AuthorMessageView
            key={message.id}
            message={message}
            pendingGhostEdit={pendingGhostEdit}
            onGhostAcceptAll={onGhostAcceptAll}
            onGhostRejectAll={onGhostRejectAll}
            onGhostAcceptHunk={onGhostAcceptHunk}
            onGhostRejectHunk={onGhostRejectHunk}
            onGhostJump={onGhostJump}
            onJumpToPath={onJumpToPath}
            onOpenDiff={onOpenDiff}
            proposalResolved={resolvedReviewMessageIds.has(
              messageUserMessageIds.get(message.id) ?? ""
            )}
            allowLiveApproval={message.id === latestActionableApprovalMessageId}
            onToolApprovalResponse={onToolApprovalResponse}
          />
        ))}
        {presetToasts.length > 0 ? (
          <AutoApplyCard
            picks={presetToasts.map(toastToPick)}
            onReviewDiff={onOpenDiff}
            onUndoAll={onPresetUndoAll}
          />
        ) : null}
        {followUpLoading || followUpChips.length > 0 ? (
          <FollowUpChips
            chips={followUpChips}
            loading={followUpLoading}
            onPick={onFollowUpPick}
          />
        ) : null}
      </ConversationContent>
      <ConversationScrollButton />
    </Conversation>
  );
}

function AuthorMessageView({
  message,
  pendingGhostEdit,
  onGhostAcceptAll,
  onGhostRejectAll,
  onGhostAcceptHunk,
  onGhostRejectHunk,
  onGhostJump,
  onJumpToPath,
  onOpenDiff,
  proposalResolved,
  allowLiveApproval,
  onToolApprovalResponse,
}: {
  message: AuthorUIMessage;
  pendingGhostEdit: GhostEdit | null;
  onGhostAcceptAll: () => void;
  onGhostRejectAll: () => void;
  onGhostAcceptHunk: (hunkId: string) => void;
  onGhostRejectHunk: (hunkId: string) => void;
  onGhostJump: (hunk: GhostHunk) => void;
  onJumpToPath?: (path: string) => void;
  onOpenDiff: () => void;
  proposalResolved: boolean;
  allowLiveApproval: boolean;
  onToolApprovalResponse: (response: {
    id: string;
    approved: boolean;
    reason?: string;
  }) => void | PromiseLike<void>;
}) {
  const text = textFromUIMessage(message);
  const plan = planForRun(message.run);
  const sources = sourcesForRun(message.run);
  const reasoning = message.parts.find((part) => part.type === "reasoning");
  const tools = message.parts.filter(
    (part): part is ToolPart => part.type === "dynamic-tool" || part.type.startsWith("tool-")
  );
  return (
    <Message from={message.role}>
      <MessageContent
        className={cn(
          "text-[13px]",
          message.role === "assistant" && "w-full max-w-full gap-3",
          message.role === "user" && "max-w-[88%] rounded-lg"
        )}
      >
        {message.role === "assistant" && plan ? <AuthorPlan plan={plan} /> : null}
        {message.role === "assistant" && reasoning ? (
          <Reasoning defaultOpen={false}>
            <ReasoningTrigger />
            <ReasoningContent>{reasoning.text}</ReasoningContent>
          </Reasoning>
        ) : null}
        {sources.length > 0 ? (
          <Sources>
            <SourcesTrigger count={sources.length} />
            <SourcesContent>
              {sources.map((source) => (
                <button
                  key={source.id}
                  type="button"
                  onClick={() => onJumpToPath?.(decodeURIComponent(source.href.slice(1)))}
                  className="flex items-center gap-2 rounded-md text-left font-mono text-[11px] text-muted-foreground transition-colors hover:text-foreground focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
                >
                  {source.title}
                </button>
              ))}
            </SourcesContent>
          </Sources>
        ) : null}
        {tools.map((part) =>
          part.type === "dynamic-tool" && part.toolName === "propose_edits" && message.run ? (
            <AuthorProposalTask
              key={part.toolCallId}
              edit={pendingGhostEdit?.runId === message.run.id ? pendingGhostEdit : null}
              run={message.run}
              onAcceptAll={onGhostAcceptAll}
              onRejectAll={onGhostRejectAll}
              onAcceptHunk={onGhostAcceptHunk}
              onRejectHunk={onGhostRejectHunk}
              onJumpToHunk={onGhostJump}
              onOpenDiff={onOpenDiff}
            />
          ) : part.type === "tool-propose_edits" ? (
            <AuthorLiveProposalTask
              key={part.toolCallId}
              part={part}
              resolved={proposalResolved || !allowLiveApproval}
              onToolApprovalResponse={onToolApprovalResponse}
            />
          ) : part.type === "tool-request_clarification" ? (
            <AuthorAskUserTask key={part.toolCallId} part={part} />
          ) : (
            <AuthorToolView key={part.toolCallId} part={part} />
          )
        )}
        {text ? <MessageResponse>{text}</MessageResponse> : null}
        {message.run?.latestOpfDocumentId ? (
          <Artifact className="bg-background/70">
            <ArtifactHeader className="px-3 py-2">
              <div className="min-w-0">
                <ArtifactTitle className="font-mono text-xs">
                  OPF version {message.run.latestOpfVersion ?? "?"}
                </ArtifactTitle>
                <ArtifactDescription className="text-[11px]">
                  {message.run.latestOpfDocumentId}
                </ArtifactDescription>
              </div>
              <ArtifactActions>
                <ArtifactAction
                  label="Open OPF version"
                  tooltip="Open the generated OPF version"
                >
                  <FileCode className="size-4" />
                </ArtifactAction>
              </ArtifactActions>
            </ArtifactHeader>
            <ArtifactContent className="p-3 text-[11px] text-muted-foreground">
              Produced by {authorOperationLabel(message.run.operation)} · {message.run.status}
            </ArtifactContent>
          </Artifact>
        ) : null}
      </MessageContent>
    </Message>
  );
}

function AuthorToolView({ part }: { part: ToolPart }) {
  const type = part.type;
  return (
    <Tool className="mb-0 bg-background/70" defaultOpen={part.state !== "output-available"}>
      {type === "dynamic-tool" ? (
        <ToolHeader type={type} state={part.state} toolName={part.toolName} />
      ) : (
        <ToolHeader type={type} state={part.state} />
      )}
      <ToolContent>
        {"input" in part ? <CompactToolInput input={part.input} /> : null}
        <AuthorToolOutput part={part} />
      </ToolContent>
    </Tool>
  );
}

function CompactToolInput({ input }: { input: unknown }) {
  const json = JSON.stringify(input ?? {}, null, 2);
  if (json.split("\n").length <= 3) {
    return (
      <Snippet code={json}>
        <SnippetInput aria-label="Tool parameters" />
        <SnippetCopyButton aria-label="Copy tool parameters" />
      </Snippet>
    );
  }
  return <ToolInput input={input} />;
}

function AuthorToolOutput({ part }: { part: ToolPart }) {
  const output = "output" in part ? part.output : undefined;
  const errorText = "errorText" in part ? part.errorText : undefined;
  if (errorText) {
    return (
      <StackTrace trace={`Error: ${errorText}`} defaultOpen={false}>
        <StackTraceHeader>
          <StackTraceError />
        </StackTraceHeader>
        <StackTraceContent />
      </StackTrace>
    );
  }
  if (isValidationOutput(output)) {
    const failed = output.errors.length + output.warnings.length;
    return (
      <TestResults
        summary={{
          failed,
          passed: output.valid ? 1 : 0,
          skipped: 0,
          total: Math.max(1, failed + (output.valid ? 1 : 0)),
        }}
      >
        <TestResultsHeader>
          <TestResultsSummary />
        </TestResultsHeader>
        <TestResultsContent>
          <TestSuite
            defaultOpen={!output.valid}
            name="OPF schema validation"
            status={output.valid ? "passed" : "failed"}
          >
            <TestSuiteName />
            {!output.valid ? (
              <div className="space-y-1 px-4 pb-3 text-[11px] text-muted-foreground">
                {[...output.errors, ...output.warnings].slice(0, 6).map((issue, index) => (
                  <div key={`${issue.path ?? "root"}-${index}`}>
                    <span className="font-mono text-foreground">{issue.path ?? "$"}</span>{" "}
                    {issue.message}
                  </div>
                ))}
              </div>
            ) : null}
          </TestSuite>
        </TestResultsContent>
      </TestResults>
    );
  }
  return <ToolOutput output={output} errorText={undefined} />;
}

function isValidationOutput(value: unknown): value is {
  valid: boolean;
  errors: Array<{ path?: string; message: string }>;
  warnings: Array<{ path?: string; message: string }>;
} {
  if (!value || typeof value !== "object" || Array.isArray(value)) return false;
  const record = value as Record<string, unknown>;
  return (
    typeof record.valid === "boolean" &&
    Array.isArray(record.errors) &&
    Array.isArray(record.warnings)
  );
}

function AuthorProposalTask({
  edit,
  run,
  onAcceptAll,
  onRejectAll,
  onAcceptHunk,
  onRejectHunk,
  onJumpToHunk,
  onOpenDiff,
}: {
  edit: GhostEdit | null;
  run: PublicAuthorRun;
  onAcceptAll: () => void;
  onRejectAll: () => void;
  onAcceptHunk: (hunkId: string) => void;
  onRejectHunk: (hunkId: string) => void;
  onJumpToHunk: (hunk: GhostHunk) => void;
  onOpenDiff: () => void;
}) {
  const proposal = run.toolMeta?.proposal;
  if (!proposal) return null;
  const pending = Boolean(edit);
  const pendingCount = edit ? pendingHunks(edit).length : 0;
  return (
    <Task className="rounded-lg border border-amber-500/30 bg-amber-500/5 p-3" defaultOpen>
      <TaskTrigger
        title={`Proposed edits · ${proposal.hunks.length} hunk${
          proposal.hunks.length === 1 ? "" : "s"
        }`}
      />
      <TaskContent className="mt-3">
        <div className="mb-3 flex flex-wrap items-center gap-2">
          <button
            type="button"
            onClick={onOpenDiff}
            className="rounded-md border border-border bg-background px-2 py-1 font-mono text-[10px] text-muted-foreground transition-colors hover:text-foreground focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
          >
            open diff
          </button>
          {pending ? (
            <>
              <button
                type="button"
                onClick={onAcceptAll}
                className="rounded-md bg-emerald-600 px-2 py-1 font-mono text-[10px] font-semibold text-white transition-colors hover:bg-emerald-700 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
              >
                approve all
              </button>
              <button
                type="button"
                onClick={onRejectAll}
                className="rounded-md border border-border bg-background px-2 py-1 font-mono text-[10px] text-muted-foreground transition-colors hover:text-foreground focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
              >
                reject all
              </button>
              <span className="font-mono text-[10px] text-muted-foreground">
                {pendingCount} pending
              </span>
            </>
          ) : null}
        </div>
        <div className="space-y-2">
          {proposal.hunks.map((hunk) => {
            const decision = edit?.decisions?.[hunk.id];
            return (
              <div key={hunk.id} className="rounded-md border border-border bg-background/70 p-2">
                <div className="mb-2 flex flex-wrap items-center justify-between gap-2">
                  <div className="min-w-0">
                    <div className="flex flex-wrap items-center gap-1.5">
                      <span className="font-mono text-[10px] uppercase text-amber-300">
                        {hunk.label}
                      </span>
                      <TaskItemFile className="font-mono">{hunk.path.replace(/^\$\.?/, "")}</TaskItemFile>
                    </div>
                    {hunk.rationale ? (
                      <p className="mt-1 text-[11px] italic text-muted-foreground">
                        {hunk.rationale}
                      </p>
                    ) : null}
                  </div>
                  {pending && !decision ? (
                    <div className="flex items-center gap-1">
                      <button
                        type="button"
                        onClick={() => onAcceptHunk(hunk.id)}
                        className="rounded-md border border-emerald-500/40 px-1.5 py-0.5 font-mono text-[10px] text-emerald-300 hover:bg-emerald-500/10 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
                      >
                        accept
                      </button>
                      <button
                        type="button"
                        onClick={() => onRejectHunk(hunk.id)}
                        className="rounded-md border border-border px-1.5 py-0.5 font-mono text-[10px] text-muted-foreground hover:text-foreground focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
                      >
                        reject
                      </button>
                    </div>
                  ) : decision ? (
                    <span className="rounded-md border border-border px-1.5 py-0.5 font-mono text-[10px] text-muted-foreground">
                      {decision}
                    </span>
                  ) : null}
                </div>
                <button
                  type="button"
                  onClick={() => onJumpToHunk(hunk)}
                  className="mb-2 font-mono text-[10px] text-muted-foreground underline-offset-2 hover:text-foreground hover:underline focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
                >
                  jump to hunk
                </button>
                <div className="grid gap-2">
                  {typeof hunk.oldValue !== "undefined" ? (
                    <CodeBlock
                      className="max-h-40 overflow-auto rounded-md border border-border/70"
                      code={JSON.stringify(hunk.oldValue, null, 2)}
                      language="json"
                    />
                  ) : null}
                  {typeof hunk.newValue !== "undefined" ? (
                    <CodeBlock
                      className="max-h-40 overflow-auto rounded-md border border-border/70"
                      code={JSON.stringify(hunk.newValue, null, 2)}
                      language="json"
                    />
                  ) : null}
                </div>
              </div>
            );
          })}
        </div>
      </TaskContent>
    </Task>
  );
}

function AuthorLiveProposalTask({
  part,
  resolved,
  onToolApprovalResponse,
}: {
  part: ToolPart;
  resolved?: boolean;
  onToolApprovalResponse: (response: {
    id: string;
    approved: boolean;
    reason?: string;
  }) => void | PromiseLike<void>;
}) {
  const input = isRecord(part.input) ? part.input : {};
  const hunks = Array.isArray(input.hunks)
    ? input.hunks.filter(isGhostHunkLike)
    : [];
  const [reviewOpen, setReviewOpen] = useState(false);
  const [selected, setSelected] = useState<Set<string>>(
    () => new Set(hunks.map((hunk) => hunk.id))
  );
  if (resolved && part.state === "approval-requested") {
    return (
      <AuthorToolView
        part={
          {
            ...part,
            state: "output-available",
            approval: undefined,
            output: {
              status: "resolved",
              summary: "Proposal already reviewed.",
            },
          } as unknown as ToolPart
        }
      />
    );
  }
  if (part.state !== "approval-requested" || !part.approval) {
    return <AuthorToolView part={part} />;
  }
  const approve = (hunkIds?: string[]) =>
    onToolApprovalResponse({
      id: part.approval.id,
      approved: true,
      reason: hunkIds ? JSON.stringify({ selectedHunkIds: hunkIds }) : undefined,
    });
  const reject = () =>
    onToolApprovalResponse({ id: part.approval.id, approved: false });
  const selectedIds = [...selected];
  return (
    <Tool className="mb-0 border-amber-500/30 bg-amber-500/5" defaultOpen>
      {part.type === "dynamic-tool" ? (
        <ToolHeader type="dynamic-tool" state={part.state} toolName={part.toolName} title="propose_edits" />
      ) : (
        <ToolHeader type={part.type} state={part.state} title="propose_edits" />
      )}
      <ToolContent>
        <div className="space-y-3">
          <div>
            <p className="font-mono text-xs text-foreground">
              {typeof input.summary === "string" ? input.summary : "Review proposed OPF edits"}
            </p>
            {typeof input.rationale === "string" ? (
              <p className="mt-1 text-[11px] text-muted-foreground">{input.rationale}</p>
            ) : null}
          </div>
          <Task defaultOpen={reviewOpen} onOpenChange={setReviewOpen}>
            <TaskTrigger title={`Review individually · ${hunks.length} hunks`} />
            <TaskContent>
              <div className="space-y-2">
                {hunks.map((hunk) => (
                  <label
                    key={hunk.id}
                    className="block rounded-md border border-border bg-background/70 p-2 text-[11px]"
                  >
                    <div className="flex items-center gap-2">
                      <input
                        type="checkbox"
                        checked={selected.has(hunk.id)}
                        onChange={(event) => {
                          setSelected((current) => {
                            const next = new Set(current);
                            if (event.target.checked) next.add(hunk.id);
                            else next.delete(hunk.id);
                            return next;
                          });
                        }}
                      />
                      <span className="font-mono text-foreground">{hunk.path}</span>
                      <span className="text-muted-foreground">{hunk.label}</span>
                    </div>
                    <div className="mt-2 grid gap-2">
                      {typeof hunk.oldValue !== "undefined" ? (
                        <CodeBlock code={JSON.stringify(hunk.oldValue, null, 2)} language="json" />
                      ) : null}
                      {typeof hunk.newValue !== "undefined" ? (
                        <CodeBlock code={JSON.stringify(hunk.newValue, null, 2)} language="json" />
                      ) : null}
                    </div>
                  </label>
                ))}
              </div>
            </TaskContent>
          </Task>
          <div className="flex flex-wrap items-center gap-2">
            <button
              type="button"
              onClick={() => approve()}
              className="rounded-md bg-emerald-600 px-2 py-1 font-mono text-[10px] font-semibold text-white hover:bg-emerald-700 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
            >
              approve all
            </button>
            <button
              type="button"
              onClick={() => approve(selectedIds)}
              className="rounded-md border border-border bg-background px-2 py-1 font-mono text-[10px] text-muted-foreground hover:text-foreground focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
            >
              approve selected ({selectedIds.length})
            </button>
            <button
              type="button"
              onClick={reject}
              className="rounded-md border border-border bg-background px-2 py-1 font-mono text-[10px] text-muted-foreground hover:text-foreground focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
            >
              reject all
            </button>
          </div>
        </div>
      </ToolContent>
    </Tool>
  );
}

function AuthorAskUserTask({ part }: { part: ToolPart }) {
  const input = isRecord(part.input) ? part.input : {};
  return (
    <Tool className="mb-0 border-brand-500/30 bg-brand-500/5" defaultOpen>
      {part.type === "dynamic-tool" ? (
        <ToolHeader type="dynamic-tool" state={part.state} toolName={part.toolName} title="request_clarification" />
      ) : (
        <ToolHeader type={part.type} state={part.state} title="request_clarification" />
      )}
      <ToolContent>
        <p className="text-sm text-foreground">
          {typeof input.question === "string" ? input.question : "Author needs clarification."}
        </p>
        {typeof input.context === "string" ? (
          <p className="text-[11px] text-muted-foreground">{input.context}</p>
        ) : null}
      </ToolContent>
    </Tool>
  );
}

function isGhostHunkLike(value: unknown): value is GhostHunk {
  return (
    isRecord(value) &&
    typeof value.id === "string" &&
    typeof value.path === "string" &&
    typeof value.label === "string" &&
    (value.op === "set" || value.op === "remove" || value.op === "rewrite")
  );
}

function AuthorToolsPane({
  catalog,
  onJumpToPath,
  onSelectTool,
  selectedToolName,
  toolCalls,
}: {
  catalog: AuthorToolCatalogItem[];
  onJumpToPath?: (path: string) => void;
  onSelectTool: (name: string) => void;
  selectedToolName: string;
  toolCalls: AuthorToolCallRecord[];
}) {
  const [kindFilter, setKindFilter] = useState<AuthorToolCatalogItem["primitive"] | "all">("all");
  const [searchQuery, setSearchQuery] = useState("");
  const selectedTool = catalog.find((tool) => tool.name === selectedToolName) ?? catalog[0] ?? null;
  const grouped = useMemo(() => {
    const query = searchQuery.trim().toLowerCase();
    const groups = new Map<string, AuthorToolCatalogItem[]>();
    for (const item of catalog) {
      if (kindFilter !== "all" && item.primitive !== kindFilter) continue;
      if (
        query &&
        ![item.name, item.title, item.ui.summary, item.description]
          .join(" ")
          .toLowerCase()
          .includes(query)
      ) {
        continue;
      }
      const group = groups.get(item.ui.group) ?? [];
      group.push(item);
      groups.set(item.ui.group, group);
    }
    const ordered = TOOL_GROUP_ORDER
      .map((group) => [group, groups.get(group) ?? []] as const)
      .filter(([, tools]) => tools.length > 0);
    const rest = [...groups.entries()].filter(
      ([group]) => !(TOOL_GROUP_ORDER as readonly string[]).includes(group)
    );
    return [...ordered, ...rest];
  }, [catalog, kindFilter, searchQuery]);
  const recentCalls = toolCalls.slice(0, 8);
  const primitiveCounts = useMemo(() => {
    const counts = new Map<AuthorToolCatalogItem["primitive"] | "all", number>([["all", catalog.length]]);
    for (const item of catalog) {
      counts.set(item.primitive, (counts.get(item.primitive) ?? 0) + 1);
    }
    return counts;
  }, [catalog]);

  return (
    <div className="flex min-h-0 flex-1 flex-col">
      <div className="border-b border-border/50 px-2 py-2">
        <label className="flex h-8 items-center gap-2 rounded-md border border-border bg-background px-2 text-[11px]">
          <Search className="size-3.5 shrink-0 text-muted-foreground" aria-hidden />
          <input
            value={searchQuery}
            onChange={(event) => setSearchQuery(event.target.value)}
            placeholder="Search tools..."
            className="min-w-0 flex-1 bg-transparent font-mono text-foreground outline-none placeholder:text-muted-foreground"
          />
          <span className="rounded-sm border border-border bg-muted/20 px-1 font-mono text-[10px] text-muted-foreground">
            ⌘P
          </span>
        </label>
        <div className="mt-2 flex flex-wrap gap-1">
          {TOOL_KIND_FILTERS.map((filter) => (
            <button
              key={filter.value}
              type="button"
              onClick={() => setKindFilter(filter.value)}
              className={cn(
                "rounded-full border px-2 py-0.5 font-mono text-[10px] transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring",
                kindFilter === filter.value
                  ? "border-brand-500/40 bg-brand-500/10 text-brand-500"
                  : "border-border bg-background text-muted-foreground hover:bg-muted/40 hover:text-foreground"
              )}
            >
              {filter.label}
              {filter.value === "resolve" ? <span className="ml-1 text-brand-500">new</span> : null}
              <span className="ml-1 text-current/60">{primitiveCounts.get(filter.value) ?? 0}</span>
            </button>
          ))}
        </div>
      </div>

      <div className="min-h-0 flex-1 overflow-y-auto px-2 py-2">
        <section className="space-y-2">
          {grouped.map(([group, tools]) => (
            <div key={group}>
              <div className="mb-1 flex items-center gap-2 px-1 font-mono text-[10px] font-semibold uppercase tracking-wider text-muted-foreground">
                {group}
                {group === "Resolve" ? (
                  <span className="rounded-sm bg-brand-500/15 px-1 text-[9px] text-brand-500">new</span>
                ) : null}
              </div>
              <ul className="space-y-1">
                {tools.map((tool) => (
                  <li key={tool.name}>
                    <ToolCatalogRow
                      selected={selectedToolName === tool.name}
                      tool={tool}
                      onSelect={() => onSelectTool(tool.name)}
                    />
                  </li>
                ))}
              </ul>
            </div>
          ))}
          {grouped.length === 0 ? (
            <div className="rounded-lg border border-dashed border-border bg-background px-3 py-4 text-center text-[11px] text-muted-foreground">
              No tools match your filters.
            </div>
          ) : null}
        </section>

        <section className="mt-4">
          <h2 className="mb-2 font-mono text-[10px] font-semibold uppercase tracking-wider text-muted-foreground">
            Recent runs
          </h2>
          {recentCalls.length > 0 ? (
            <ul className="space-y-1.5">
              {recentCalls.map((call) => (
                <li key={call.id} className="rounded-lg border border-border bg-background px-2 py-2">
                  <div className="mb-1 flex items-center justify-between gap-2">
                    <span className="font-mono text-[11px] text-foreground">{call.name}</span>
                    <span className="rounded-md bg-muted/30 px-1.5 py-0.5 font-mono text-[10px] text-muted-foreground">
                      {call.status}
                    </span>
                  </div>
                  <p className="text-[11px] leading-snug text-muted-foreground">
                    {call.outputSummary ?? call.error ?? "No output summary."}
                  </p>
                  {call.affectedPaths?.length ? (
                    <div className="mt-2 flex flex-wrap gap-1">
                      {call.affectedPaths.slice(0, 4).map((path) => (
                        <button
                          key={path}
                          type="button"
                          onClick={() => onJumpToPath?.(path)}
                          className="rounded-md border border-border px-1.5 py-0.5 font-mono text-[10px] text-muted-foreground transition-colors hover:border-brand-500/50 hover:text-foreground focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
                        >
                          {path.replace(/^\$\.?/, "")}
                        </button>
                      ))}
                    </div>
                  ) : null}
                </li>
              ))}
            </ul>
          ) : (
            <div className="rounded-lg border border-dashed border-border bg-background px-3 py-4 text-center text-[11px] text-muted-foreground">
              Ask Author for an edit or run a manual tool; calls will appear here.
            </div>
          )}
        </section>
      </div>
    </div>
  );
}

  tool: AuthorToolCatalogItem;
}) {
  return (
    <button
      type="button"
      onClick={onSelect}
      className={cn(

      )}
    >
      <div className="flex items-start gap-2">
        <ChevronRight className={cn("mt-0.5 size-3 shrink-0 text-muted-foreground", selected && "rotate-90 text-brand-500")} />
        <div className="min-w-0 flex-1">
          <div className="flex items-center gap-1.5">
            <span className="truncate font-mono text-[11px] text-foreground">{tool.name}</span>
            {isNewTool(tool.name) ? (
              <span className="rounded-sm bg-brand-500/15 px-1 font-mono text-[9px] uppercase text-brand-500">
                new
              </span>
            ) : null}
          </div>
          <p className="mt-0.5 truncate text-[11px] text-muted-foreground">{tool.ui.summary}</p>
          {selected ? (
            <div className="mt-2 space-y-1.5">
              <p className="text-[11px] leading-snug text-muted-foreground">{tool.description}</p>
              <div className="flex flex-wrap items-center gap-1.5">
                <ToolTonePill tool={tool} />
                <span className="rounded-md border border-border bg-muted/20 px-1.5 py-0.5 font-mono text-[10px] text-muted-foreground">
                  {tool.inputSchema}
                </span>
              </div>
            </div>
          ) : null}
        </div>
        {!selected ? <ToolTonePill tool={tool} /> : null}
      </div>
    </button>
  );
}

function ToolTonePill({ tool }: { tool: AuthorToolCatalogItem }) {
  return (
    <span
      className={cn(
        "shrink-0 rounded-md border px-1.5 py-0.5 font-mono text-[10px]",
        tool.primitive === "resolve"
          ? "border-brand-500/40 text-brand-500"
          : tool.readOnly
            ? "border-border text-muted-foreground"
            : tool.requiresReview
              ? "border-amber-500/40 text-amber-300"
              : "border-brand-500/40 text-brand-500"
      )}
    >
      {tool.primitive === "resolve" ? "resolve" : tool.readOnly ? "read" : tool.requiresReview ? "review" : "create"}
    </span>
  );
}

function isNewTool(name: string): boolean {
  return name === "resolve.field" || name === "create.user_preference";
}

function AuthorGalleryRail({
  document,
  gallerySources,
  manualKind,
  onActionStateChange,
  onClearManual,
  onClearPreview,
  onJumpToPath,
  onManualKind,
  onPreview,
  onTogglePin,
  pinnedPath,
  pinnedPaths,
  preview,
  selectedPath,
}: {
  document: OPFDocument | null;
  gallerySources: ReadonlyArray<AuthorGallerySource>;
  manualKind: GalleryKind | null;
  onActionStateChange: (state: GalleryActionState) => void;
  onClearManual: () => void;
  onClearPreview: () => void;
  onJumpToPath: (path: string) => void;
  onManualKind: (kind: GalleryKind) => void;
  onPreview: (preview: GalleryPreview | null) => void;
  onTogglePin: (path: string) => void;
  pinnedPath: string | null;
  pinnedPaths: ReadonlyArray<string>;
  preview: GalleryPreview | null;
  selectedPath: string | null;
}) {
  const [expandedKinds, setExpandedKinds] = useState<GalleryKind[]>([]);
  const [searchOpen, setSearchOpen] = useState(false);
  const [search, setSearch] = useState("");
  const [chosen, setChosen] = useState<{ contextKey: string; id: string } | null>(null);
  const bodyRef = useRef<HTMLDivElement | null>(null);

  const cursorContext = useMemo(
    () => galleryContextForPath(document, selectedPath, "selection"),
    [document, selectedPath]
  );
  const pinnedContext = useMemo(
    () => galleryContextForPath(document, pinnedPath, "pin"),
    [document, pinnedPath]
  );
  const manualContext = useMemo(
    () =>
      manualKind
        ? manualGalleryContext(manualKind, document, selectedPath, pinnedContext ?? cursorContext)
        : null,
    [cursorContext, document, manualKind, pinnedContext, selectedPath]
  );
  const activeContext = manualContext ?? pinnedContext ?? cursorContext;
  const usage = useMemo(() => galleryUsage(document), [document]);
  const contextItems = useMemo(
    () => (activeContext ? galleryItemsForContext(activeContext, document, search, gallerySources) : []),
    [activeContext, document, gallerySources, search]
  );
  const activeKey = activeContext ? galleryContextKey(activeContext) : "document";
  const chosenId =
    chosen?.contextKey === activeKey
      ? chosen.id
      : activeContext?.currentId ?? contextItems[0]?.id ?? null;

  const chosenItem =
    contextItems.find((item) => item.id === chosenId) ??
    contextItems.find((item) => item.id === activeContext?.currentId) ??
    contextItems[0] ??
    null;

  useEffect(() => {
    bodyRef.current?.scrollTo({ top: 0 });
  }, [activeKey]);

  useEffect(() => {
    onActionStateChange({ context: activeContext, item: chosenItem });
  }, [activeContext, chosenItem, onActionStateChange]);

  const toggleExpanded = (kind: GalleryKind) => {
    setExpandedKinds((current) =>
      current.includes(kind) ? current.filter((item) => item !== kind) : [...current, kind]
    );
  };

  const previewItem = (path: string | null, item: GalleryItem | null) => {
    if (!path || !item) return;
    onPreview({ path, item });
  };

  const chooseGalleryItem = (kind: GalleryKind, id: string) => {
    const context =
      activeContext?.kind === kind
        ? activeContext
        : manualGalleryContext(kind, document, selectedPath, pinnedContext ?? cursorContext);
    setChosen({ contextKey: galleryContextKey(context), id });
    if (activeContext?.kind !== kind) onManualKind(kind);
  };

  return (
    <div data-testid="author-gallery-rail" className="flex min-h-0 flex-1 flex-col bg-background">
      <div ref={bodyRef} className="min-h-0 flex-1 overflow-y-auto">
        {activeContext ? (
          <GalleryContextBlock
            activeItem={chosenItem}
            context={activeContext}
            contextItems={contextItems}
            onChoose={(id) => chooseGalleryItem(activeContext.kind, id)}
            onClearManual={onClearManual}
            onClearPreview={onClearPreview}
            onPreview={previewItem}
            onTogglePin={onTogglePin}
            pinned={Boolean(activeContext.path && pinnedPaths.includes(activeContext.path))}
            preview={preview}
            usage={usage}
          />
        ) : (
          <GalleryDocumentBlock
            document={document}
            onBrowse={(kind) => {
              onManualKind(kind);
              if (!expandedKinds.includes(kind)) setExpandedKinds((current) => [...current, kind]);
            }}
            onJumpToPath={onJumpToPath}
            usage={usage}
            gallerySources={gallerySources}
          />
        )}

        <div className="flex h-8 items-center justify-between border-y border-border/60 bg-muted/20 px-3">
          <div className="font-mono text-[11px] font-semibold uppercase tracking-wider text-muted-foreground">
            Browse
          </div>
          {searchOpen ? (
            <label className="flex min-w-0 flex-1 items-center gap-1.5 pl-3">
              <Search className="size-3 text-muted-foreground" aria-hidden />
              <input
                autoFocus
                value={search}
                onChange={(event) => setSearch(event.target.value)}
                onKeyDown={(event) => {
                  if (event.key === "Escape") {
                    setSearch("");
                    setSearchOpen(false);
                  }
                }}
                placeholder="Search gallery"
                className="h-6 min-w-0 flex-1 bg-transparent font-mono text-[11px] text-foreground outline-none placeholder:text-muted-foreground"
              />
            </label>
          ) : (
            <button
              type="button"
              onClick={() => setSearchOpen(true)}
              className="inline-flex size-6 items-center justify-center rounded-md text-muted-foreground transition-colors hover:bg-muted/60 hover:text-foreground focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
              aria-label="Search gallery"
            >
              <Search className="size-3.5" aria-hidden />
            </button>
          )}
        </div>

        <GalleryCatalogTree
          activeContext={activeContext}
          document={document}
          expandedKinds={expandedKinds}
          onClearPreview={onClearPreview}
          onChoose={chooseGalleryItem}
          onManualKind={onManualKind}
          onPreview={previewItem}
          onToggleExpanded={toggleExpanded}
          preview={preview}
          search={search}
          selectedId={chosen?.contextKey === activeKey ? chosen.id : chosenId}
          usage={usage}
          gallerySources={gallerySources}
        />
      </div>
    </div>
  );
}

function GalleryContextBlock({
  activeItem,
  context,
  contextItems,
  onChoose,
  onClearManual,
  onClearPreview,
  onPreview,
  onTogglePin,
  pinned,
  preview,
  usage,
}: {
  activeItem: GalleryItem | null;
  context: GalleryContext;
  contextItems: GalleryItem[];
  onChoose: (id: string) => void;
  onClearManual: () => void;
  onClearPreview: () => void;
  onPreview: (path: string | null, item: GalleryItem | null) => void;
  onTogglePin: (path: string) => void;
  pinned: boolean;
  preview: GalleryPreview | null;
  usage: GalleryUsage;
}) {
  const meta = GALLERY_META[context.kind];
  const contextTitle = context.source === "manual" ? "BROWSE CATALOG" : "FOR YOUR SELECTION";

  return (
    <section
      className={cn(
        "border-b border-border bg-card/20",
        context.source === "manual" ? "bg-muted/10" : "border-b-brand-500/50",
        pinned && "ring-1 ring-inset ring-brand-500/35"
      )}
    >
      <header className="border-b border-dashed border-brand-500/40 px-3 py-3">
        <div className="flex items-start justify-between gap-2">
          <div className="min-w-0">
            <div className="font-mono text-[11px] font-semibold uppercase tracking-wider text-brand-500">
              {contextTitle}
            </div>
            <div className="mt-1 flex min-w-0 items-center gap-1.5 font-mono text-sm text-foreground">
              <span className="text-muted-foreground">{meta.glyph}</span>
              <span>{meta.label}</span>
              <span className="text-muted-foreground">·</span>
              <span className="truncate text-muted-foreground">{context.scope}</span>
            </div>
            <div className="mt-1 truncate font-mono text-xs text-muted-foreground">
              {context.path ? (
                <>
                  currently <span className="text-foreground">{context.currentLabel}</span>
                </>
              ) : (
                "manual catalog override"
              )}
            </div>
            {pinned && context.path ? (
              <div className="mt-2 inline-flex max-w-full items-center gap-1 rounded-md border border-brand-500/35 bg-brand-500/10 px-1.5 py-0.5 font-mono text-[11px] text-brand-500">
                <Pin className="size-3 shrink-0" aria-hidden />
                <span className="truncate">{context.path.replace(/^\$\.?/, "")}</span>
              </div>
            ) : null}
          </div>
          {context.path ? (
            <button
              type="button"
              aria-pressed={pinned}
              onClick={() => onTogglePin(context.path as string)}
              className={cn(
                "inline-flex min-h-9 shrink-0 flex-col items-center justify-center rounded-md border px-2 font-mono text-[10px] uppercase tracking-wide transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring",
                pinned
                  ? "border-brand-500/50 bg-brand-500/15 text-brand-500"
                  : "border-border bg-background text-muted-foreground hover:bg-muted/40 hover:text-foreground"
              )}
            >
              <Pin className="size-3.5" aria-hidden />
              Pin
            </button>
          ) : (
            <button
              type="button"
              onClick={onClearManual}
              className="inline-flex h-8 shrink-0 items-center rounded-md border border-border bg-background px-2 font-mono text-[11px] text-muted-foreground hover:bg-muted/40 hover:text-foreground focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
            >
              cursor →
            </button>
          )}
        </div>
      </header>

      <div className="px-3 py-3">
        {contextItems.length > 0 ? (
          <div className={cn("overflow-y-auto pr-1", contextItems.length > 6 && "max-h-[440px]")}>
            <GalleryGrid
              activeId={context.currentId}
              items={contextItems.slice(0, context.source === "manual" ? 12 : 8)}
              onChoose={onChoose}
              onClearPreview={onClearPreview}
              onPreview={(item) => onPreview(context.path, item)}
              path={context.path}
              preview={preview}
              selectedId={activeItem?.id ?? null}
              usage={usage}
            />
          </div>
        ) : (
          <div className="rounded-lg border border-dashed border-border bg-background/60 px-3 py-5 text-center text-xs text-muted-foreground">
            No compatible gallery items found.
          </div>
        )}
      </div>
    </section>
  );
}

function GalleryDocumentBlock({
  document,
  gallerySources,
  onBrowse,
  onJumpToPath,
  usage,
}: {
  document: OPFDocument | null;
  gallerySources: ReadonlyArray<AuthorGallerySource>;
  onBrowse: (kind: GalleryKind) => void;
  onJumpToPath: (path: string) => void;
  usage: GalleryUsage;
}) {
  const refs = useMemo(() => galleryDocumentRefs(document, usage, gallerySources), [document, gallerySources, usage]);
  const refsByKind = GALLERY_KINDS.map((kind) => ({
    kind,
    refs: refs.filter((ref) => ref.kind === kind),
  })).filter((group) => group.refs.length > 0);
  const catalogCount = refsByKind.length;

  return (
    <section className="border-b border-border bg-card/10 px-3 py-3">
      <header className="mb-3">
        <div className="font-mono text-[11px] font-semibold uppercase tracking-wider text-muted-foreground">
          In this document
        </div>
        <div className="mt-1 font-mono text-sm text-foreground">
          {refs.length} reference{refs.length === 1 ? "" : "s"} · {catalogCount} catalog{catalogCount === 1 ? "" : "s"}
        </div>
      </header>
      {refsByKind.length > 0 ? (
        <div className="space-y-4">
          {refsByKind.map((group) => (
            <div key={group.kind}>
              <div className="mb-2 flex items-center justify-between gap-2 font-mono text-xs">
                <span className="text-muted-foreground">
                  {GALLERY_META[group.kind].label} · {galleryKindUseCount(usage, group.kind)}
                </span>
                <button
                  type="button"
                  onClick={() => onBrowse(group.kind)}
                  className="text-muted-foreground transition-colors hover:text-foreground focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
                >
                  browse →
                </button>
              </div>
              <div className="grid grid-cols-2 gap-2">
                {group.refs.slice(0, 4).map((ref) => (
                  <button
                    key={`${ref.kind}-${ref.id}-${ref.path}`}
                    type="button"
                    onClick={() => onJumpToPath(ref.path)}
                    className="min-w-0 rounded-lg border border-border bg-background text-left transition-colors hover:border-brand-500/55 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
                    title={ref.path}
                  >
                    <GalleryThumbnail item={ref.item} inUseCount={ref.count} />
                    <GalleryCardFooter item={ref.item} />
                  </button>
                ))}
              </div>
            </div>
          ))}
        </div>
      ) : (
        <div className="rounded-lg border border-dashed border-border bg-background/60 px-3 py-5 text-center text-xs text-muted-foreground">
          No gallery references in this OPF yet.
        </div>
      )}
    </section>
  );
}

function GalleryCatalogTree({
  activeContext,
  document,
  expandedKinds,
  gallerySources,
  onClearPreview,
  onChoose,
  onManualKind,
  onPreview,
  onToggleExpanded,
  preview,
  search,
  selectedId,
  usage,
}: {
  activeContext: GalleryContext | null;
  document: OPFDocument | null;
  expandedKinds: ReadonlyArray<GalleryKind>;
  gallerySources: ReadonlyArray<AuthorGallerySource>;
  onClearPreview: () => void;
  onChoose: (kind: GalleryKind, id: string) => void;
  onManualKind: (kind: GalleryKind) => void;
  onPreview: (path: string | null, item: GalleryItem | null) => void;
  onToggleExpanded: (kind: GalleryKind) => void;
  preview: GalleryPreview | null;
  search: string;
  selectedId: string | null;
  usage: GalleryUsage;
}) {
  return (
    <section className="px-3 py-2">
      <div className="space-y-3">
        {GALLERY_GROUPS.map((group) => (
          <div key={group.id}>
            <div className="mb-1 px-2 font-mono text-[10px] font-semibold uppercase tracking-wider text-muted-foreground">
              {group.label}
            </div>
            <div className="space-y-1">
        {group.kinds.map((kind) => {
          const expanded = expandedKinds.includes(kind);
          const active = activeContext?.kind === kind;
          const items = filterGalleryItems(galleryItemsForKind(kind, document, gallerySources), search);
          const inUse = galleryKindUseCount(usage, kind);
          const path = active ? activeContext?.path ?? null : null;

          return (
            <div
              key={kind}
              className={cn(
                "rounded-lg border border-transparent",
                active && "border-brand-500/40 bg-brand-500/10"
              )}
            >
              <button
                type="button"
                onClick={() => {
                  onManualKind(kind);
                  onToggleExpanded(kind);
                }}
                className="flex h-10 w-full items-center gap-2 px-2 text-left font-mono text-xs text-muted-foreground transition-colors hover:text-foreground focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
              >
                {expanded ? <ChevronDown className="size-3.5" /> : <ChevronRight className="size-3.5" />}
                <span className="w-5 text-center text-foreground">{GALLERY_META[kind].glyph}</span>
                <span className="min-w-0 flex-1 truncate">{GALLERY_META[kind].label}</span>
                {inUse > 0 ? (
                  <span className="rounded-full bg-brand-500/15 px-1.5 py-0.5 text-[10px] text-brand-500">
                    ×{inUse}
                  </span>
                ) : null}
              </button>
              {expanded ? (
                <div className="grid grid-cols-2 gap-2 px-2 pb-2">
                  {items.slice(0, 12).map((item) => (
                    <GalleryItemCard
                      key={`${kind}-${item.id}`}
                      active={activeContext?.currentId === item.id}
                      inUseCount={galleryItemUseCount(usage, kind, item.id)}
                      item={item}
                      onChoose={() => onChoose(kind, item.id)}
                      onClearPreview={onClearPreview}
                      onPreview={path ? () => onPreview(path, item) : undefined}
                      path={path}
                      previewed={preview?.item.id === item.id && preview.path === path}
                      selected={activeContext?.kind === kind && selectedId === item.id}
                    />
                  ))}
                </div>
              ) : null}
            </div>
          );
        })}
            </div>
          </div>
        ))}
      </div>
    </section>
  );
}

function AgentActionDock({
  attachments,
  centerTab,
  chatStatus,
  fileInputRef,
  isSending,
  onStop,
  onDropFiles,
  onFileInputChange,
  onOperationModeChange,
  onPromptChange,
  onRemoveAttachment,
  onSample,
  onSignIn,
  onSubmit,
  operationMode,
  prompt,
  previewSlideContext,
  localFieldUpdateActive,
  signedIn,
  thread,
  cursorPath,
  pinnedPaths,
  onUnpinPath,
  chipSuggestions,
  hasMessages,
  intent,
  presetMatches,
  onPresetApply,
}: {
  attachments: Attachment[];
  centerTab: CenterTabId;
  chatStatus: "submitted" | "streaming" | "ready" | "error";
  fileInputRef: RefObject<HTMLInputElement | null>;
  isSending: boolean;
  onStop: () => void;
  onDropFiles: (files: FileList | File[]) => void;
  onFileInputChange: (event: ChangeEvent<HTMLInputElement>) => void;
  onOperationModeChange: (mode: OperationMode) => void;
  onPromptChange: (value: string) => void;
  onRemoveAttachment: (id: string) => void;
  onSample: () => void;
  onSignIn: () => void;
  onSubmit: () => void;
  operationMode: OperationMode;
  prompt: string;
  previewSlideContext: PreviewSlideContext | null;
  localFieldUpdateActive: boolean;
  signedIn: boolean;
  thread: PublicAuthorThread | null;
  cursorPath: string | null;
  pinnedPaths: ReadonlyArray<string>;
  onUnpinPath: (path: string) => void;
  chipSuggestions: ReadonlyArray<ChipSuggestion>;
  hasMessages: boolean;
  intent: ReturnType<typeof deriveAuthorIntent>;
  presetMatches: ReturnType<typeof matchGalleryPresets>;
  onPresetApply: (suggestion: PresetSuggestion) => void;
}) {
  const handleDrop = (event: DragEvent<HTMLFormElement>) => {
    event.preventDefault();
    if (event.dataTransfer.files.length > 0) onDropFiles(event.dataTransfer.files);
  };
  const guestAiAction = !signedIn && !thread && !localFieldUpdateActive;
  const submitLabel = localFieldUpdateActive
    ? "Update field"
    : guestAiAction
      ? "Sign in"
      : "Go";

  const trimmedCursor = cursorPath ? cursorPath.replace(/^\$\.?/, "") : "";
  const previewContextActive = centerTab === "preview" && previewSlideContext;
  const hasContext = pinnedPaths.length > 0 || Boolean(previewContextActive) || Boolean(trimmedCursor);

  const emptyState = !thread && !hasMessages;
  const effectiveChips: ReadonlyArray<ChipSuggestion> = emptyState
    ? [{ id: "sample", label: "try a sample", prompt: "" }, ...chipSuggestions]
    : chipSuggestions;
  const attachmentData = attachments.map((attachment) => ({
    id: attachment.id,
    type: "file" as const,
    mediaType: attachment.type || "application/octet-stream",
    filename: attachment.name,
    url: "#",
  }));

  return (
    <div>
      <IntentDossierStrip
        intent={intent}
        className="mb-1.5 rounded-md border border-border/40"
      />
      {/*
       * Show the clarifying card when intent isn't rich AND a critical
       * slot is missing AND the user hasn't started the conversation.
       * Triggering only on `vagueness === "vague"` would mean every
       * variant of the card except audience was unreachable, because
       * "vague" implies zero slots filled and `clarifyingKindFor` would
       * always return "audience" in that state.
       */}
      {intent.vagueness !== "rich" &&
      !hasMessages &&
      (!intent.audience || !intent.tone || !intent.topic) ? (
        <ClarifyingCard
          kind={clarifyingKindFor({
            audience: intent.audience?.value,
            tone: intent.tone?.value,
            topic: intent.topic?.value,
          })}
          onPick={(value, kind) => {
            const phrase =
              kind === "audience"
                ? `for ${value.replace(/-/g, " ")}`
                : kind === "tone"
                  ? `in a ${value} tone`
                  : value;
            const next = prompt.trim() ? `${prompt.trimEnd()} ${phrase}` : phrase;
            onPromptChange(next);
          }}
          onSkip={onSubmit}
          className="mb-1.5"
        />
      ) : null}
      {thread || hasContext ? (
        <div className="mb-1.5 flex items-center gap-1.5">
          {thread ? (
            <div className="relative">
              <select
                aria-label="Operation mode"
                value={operationMode}
                onChange={(event) => onOperationModeChange(event.target.value as OperationMode)}
                className="h-6 appearance-none rounded-md border border-border bg-background pl-2 pr-6 font-mono text-[11px] text-muted-foreground transition-colors hover:text-foreground focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
              >
                {OPERATION_MODE_OPTIONS.map((mode) => (
                  <option key={mode.value} value={mode.value}>{mode.label}</option>
                ))}
              </select>
              <ChevronDown className="pointer-events-none absolute right-1.5 top-1/2 size-3 -translate-y-1/2 text-muted-foreground" aria-hidden />
            </div>
          ) : null}

          <div className="flex min-w-0 flex-1 flex-wrap items-center gap-1 text-[11px]">
            {pinnedPaths.length > 0
              ? pinnedPaths.map((p) => (
                  <span
                    key={`pin-${p}`}
                    className="inline-flex items-center gap-1 rounded-md border border-brand-500/40 bg-brand-500/10 px-1.5 py-0.5 font-mono text-[10px] text-brand-500"
                    title="Pinned: included as LLM context"
                  >
                    {p.replace(/^\$\.?/, "")}
                    <button
                      type="button"
                      onClick={() => onUnpinPath(p)}
                      className="ml-0.5 inline-flex size-3 items-center justify-center rounded-sm text-brand-500/70 hover:bg-brand-500/20 hover:text-brand-500 focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-ring"
                      aria-label={`Unpin ${p}`}
                    >
                      ×
                    </button>
                  </span>
                ))
              : previewContextActive
                ? (
                    <span className="inline-flex items-center gap-1 rounded-md border border-brand-500/40 bg-brand-500/10 px-1.5 py-0.5 font-mono text-[10px] text-brand-500">
                      {previewContextActive.label}
                    </span>
                  )
                : trimmedCursor
                  ? (
                      <span className="inline-flex items-center rounded-md border border-border bg-background px-1.5 py-0.5 font-mono text-[10px] text-muted-foreground">
                        {trimmedCursor}
                      </span>
                    )
                  : null}
          </div>
        </div>
      ) : null}

      <PromptInput
        accept={ACCEPTED_UPLOAD_EXTENSIONS.join(",")}
        aria-label="Agent message composer"
        className="rounded-lg border border-input bg-background focus-within:ring-2 focus-within:ring-ring hover:border-foreground/20"
        multiple
        onDragOver={(event) => event.preventDefault()}
        onDrop={handleDrop}
        onSubmit={() => {
          if (guestAiAction) {
            onSignIn();
            return;
          }
          onSubmit();
        }}
      >
        <input
          ref={fileInputRef}
          type="file"
          className="sr-only"
          multiple
          accept={ACCEPTED_UPLOAD_EXTENSIONS.join(",")}
          onChange={onFileInputChange}
        />
        <PromptInputBody>
          <PromptInputTextarea
            aria-label="Agent message"
            className="min-h-[3.5rem] font-mono text-sm"
            disabled={isSending}
            onChange={(event) => onPromptChange(event.target.value)}
            placeholder={
              thread
                ? "Describe the edit, or drop a file..."
                : signedIn
                  ? "Describe a deck, paste OPF, or drop a file..."
                  : "Paste OPF JSON, or sign in for AI..."
            }
            value={prompt}
          />
        </PromptInputBody>
        <PromptInputFooter className="p-1.5">
          <PromptInputTools>
            <PromptInputButton
              aria-label="Attach source files"
              onClick={() => fileInputRef.current?.click()}
              type="button"
              variant="ghost"
            >
              <Paperclip className="size-3.5" aria-hidden />
              <span className="sr-only">Attach</span>
            </PromptInputButton>
          </PromptInputTools>
          <PromptInputSubmit
            disabled={isSending && chatStatus !== "streaming"}
            onStop={onStop}
            status={chatStatus === "streaming" ? "streaming" : isSending ? "submitted" : "ready"}
          >
            {isSending ? <Loader2 className="size-3 animate-spin" /> : <Play className="size-3" />}
            <span className="sr-only">{submitLabel}</span>
          </PromptInputSubmit>
        </PromptInputFooter>
      </PromptInput>

      {attachments.length > 0 ? (
        <Attachments className="mt-1.5" variant="inline">
          {attachmentData.map((attachment) => (
            <Attachment
              key={attachment.id}
              data={attachment}
              onRemove={() => onRemoveAttachment(attachment.id)}
            >
              <AttachmentPreview />
              <AttachmentInfo />
              <AttachmentRemove />
            </Attachment>
          ))}
        </Attachments>
      ) : null}

      <div className="mt-1.5 flex items-center gap-1.5">
        <div className="flex min-w-0 flex-1 flex-wrap items-center gap-1">
          {effectiveChips.map((chip) => (
            <Suggestion
              key={chip.id}
              onClick={() => {
                if (chip.id === "sample") {
                  onSample();
                  return;
                }
                const next = prompt.trim() ? `${prompt.trimEnd()} ${chip.prompt}` : chip.prompt;
                onPromptChange(next);
              }}
              className="h-6 rounded-md px-1.5 font-mono text-[10px]"
              suggestion={chip.label}
              title={chip.prompt || chip.label}
            >
              {chip.label}
            </Suggestion>
          ))}
        </div>
      </div>
      <PresetSuggestionStrip
        matches={presetMatches}
        onApply={onPresetApply}
        className="mt-1.5 rounded-md border border-border/40"
      />
    </div>
  );
}

function ToolsActionDock({
  attachments,
  catalog,
  document,
  gallerySources,
  manualOutput,
  onManualOutput,
  onRecordToolCall,
  onSelectTool,
  selectedPath,
  selectedToolName,
}: {
  attachments: Attachment[];
  catalog: AuthorToolCatalogItem[];
  document: OPFDocument | null;
  gallerySources: ReadonlyArray<AuthorGallerySource>;
  manualOutput: ManualToolOutput;
  onManualOutput: (output: ManualToolOutput) => void;
  onRecordToolCall: (call: AuthorToolCallRecord) => void;
  onSelectTool: (name: string) => void;
  selectedPath: string | null;
  selectedToolName: string;
}) {
  const [query, setQuery] = useState("");
  const [kind, setKind] = useState<GalleryKind | "all">("all");
  const [advancedOpen, setAdvancedOpen] = useState(false);
  const [advancedJson, setAdvancedJson] = useState("{}");
  const [resolvePath, setResolvePath] = useState(selectedPath ?? "narrative");
  const [resolveLayers, setResolveLayers] = useState<"all" | "winning-only">("all");
  const [copiedManualJson, setCopiedManualJson] = useState(false);
  const selectedTool = catalog.find((tool) => tool.name === selectedToolName) ?? catalog[0] ?? null;
  const runnable = selectedTool ? manualToolCanRun(selectedTool.name, document) : false;
  const disabledReason = selectedTool ? manualToolDisabledReason(selectedTool, document) : "No tool selected.";
  const manualOutputJson = manualOutput ? formatValue(manualOutput.result) : "";

  useEffect(() => {
    if (selectedToolName === "resolve.field" && selectedPath) {
      // eslint-disable-next-line react-hooks/set-state-in-effect
      setResolvePath(selectedPath.replace(/^\$\.?/, ""));
    }
  }, [selectedPath, selectedToolName]);

  const runTool = () => {
    if (!selectedTool || !runnable) return;
    const definition = authorToolDefinition(selectedTool.name);
    if (!definition) return;

    let input: unknown;
    try {
      input = advancedOpen && advancedJson.trim() !== "{}"
        ? JSON.parse(advancedJson)
        : manualToolInput(selectedTool.name, {
            attachments,
            document,
            gallerySources,
            kind,
            query,
            resolveLayers,
            resolvePath,
            selectedPath,
          });
    } catch (error) {
      onManualOutput({
        name: selectedTool.name,
        result: { error: error instanceof Error ? error.message : String(error) },
        ranAt: new Date().toISOString(),
      });
      return;
    }

    let result: unknown;
    const call = recordAuthorToolCall(definition, input, () => {
      result = executeStaticAuthorTool({ name: selectedTool.name, input });
      return manualToolCallSummary(selectedTool.name, result);
    });
    onRecordToolCall(call);
    onManualOutput({
      name: selectedTool.name,
      result: call.status === "success" ? result : { error: call.error },
      ranAt: new Date().toISOString(),
    });
  };

  const copyManualJson = () => {
    if (!manualOutputJson) return;
    void navigator.clipboard
      .writeText(manualOutputJson)
      .then(() => {
        setCopiedManualJson(true);
        window.setTimeout(() => setCopiedManualJson(false), 1200);
      })
      .catch(() => {
        // Clipboard writes can fail when the document is not focused (e.g. browser automation).
        setCopiedManualJson(false);
      });
  };

  return (
    <div className="space-y-3">
      <div className="flex items-start justify-between gap-3">
        <div className="min-w-0">
          <div className="font-mono text-[10px] font-semibold uppercase tracking-wider text-muted-foreground">
            Run {selectedTool?.name ?? "tool"}
          </div>
          <p className="mt-1 truncate text-[11px] text-muted-foreground">
            {runnable ? selectedTool?.ui.summary : disabledReason}
          </p>
        </div>
        <div className="inline-flex shrink-0 rounded-md border border-border bg-background p-0.5 text-[10px]">
          <span className="rounded-sm px-1.5 py-0.5 text-muted-foreground">Auto</span>
          <span className="rounded-sm bg-brand-500/10 px-1.5 py-0.5 font-mono text-brand-500">Manual</span>
        </div>
      </div>

      {selectedToolName === "resolve.field" ? (
        <div className="grid gap-2">
          <label className="grid gap-1">
            <span className="font-mono text-[10px] text-muted-foreground">path</span>
            <input
              value={resolvePath}
              onChange={(event) => setResolvePath(event.target.value)}
              placeholder="narrative"
              className="h-9 rounded-md border border-border bg-background px-2.5 font-mono text-xs text-foreground outline-none placeholder:text-muted-foreground focus:border-brand-500/70"
            />
          </label>
          <label className="grid gap-1">
            <span className="font-mono text-[10px] text-muted-foreground">layers</span>
            <select
              value={resolveLayers}
              onChange={(event) => setResolveLayers(event.target.value as "all" | "winning-only")}
              className="h-9 rounded-md border border-border bg-background px-2 font-mono text-xs text-foreground outline-none focus:border-brand-500/70"
            >
              <option value="all">all</option>
              <option value="winning-only">winning-only</option>
            </select>
          </label>
        </div>
      ) : selectedToolName === "search.gallery" ? (
        <div className="grid grid-cols-[minmax(0,1fr)_auto] gap-2">
          <input
            value={query}
            onChange={(event) => setQuery(event.target.value)}
            placeholder="Search gallery"
            className="h-9 min-w-0 rounded-md border border-border bg-background px-2.5 font-mono text-xs text-foreground outline-none placeholder:text-muted-foreground focus:border-brand-500/70"
          />
          <select
            value={kind}
            onChange={(event) => setKind(event.target.value as GalleryKind | "all")}
            className="h-9 rounded-md border border-border bg-background px-2 font-mono text-xs text-foreground outline-none focus:border-brand-500/70"
          >
            <option value="all">all</option>
            {GALLERY_KINDS.map((item) => (
              <option key={item} value={item}>
                {item}
              </option>
            ))}
          </select>
        </div>
      ) : null}

      <details open={advancedOpen} onToggle={(event) => setAdvancedOpen(event.currentTarget.open)}>
        <summary className="cursor-pointer font-mono text-[11px] text-muted-foreground">
          advanced JSON
        </summary>
        <textarea
          value={advancedJson}
          onChange={(event) => setAdvancedJson(event.target.value)}

function SchemaActionDock({
  document,
  gallerySources,
  isSending,
  onApplyValue,
  selectedPath,
  validation,
}: {
  document: OPFDocument | null;
  gallerySources: ReadonlyArray<AuthorGallerySource>;
  isSending: boolean;
  onApplyValue: (path: string, value: unknown) => void;
  selectedPath: string | null;
  validation: ReturnType<typeof validateOPF> | null;
}) {
  const path = selectedPath ? normalizeSchemaPath(selectedPath) : null;
  const field = path ? findSchemaField(path) : null;
  const currentValue = document && path ? readValueAtPath(document, path) : undefined;
  const preview = path ? schemaValuePreviewForPath(document, path) : null;
  const proposal = field && path ? proposalForUnsetPath(field, document, path) : null;
  const galleryContext = galleryContextForPath(document, path, "selection");
  const galleryItems = galleryContext ? galleryItemsForContext(galleryContext, document, "", gallerySources) : [];
  const enumOptions = schemaChoiceOptions(path, field?.type);
  const editorMode = schemaEditorMode({ currentValue, fieldType: field?.type, galleryContext, enumOptions });
  const initialText = schemaEditorInitialText(currentValue, proposal?.value, editorMode);
  const draftKey = `${path ?? "none"}:${editorMode}:${initialText}`;
  const [draftState, setDraftState] = useState<{
    key: string;
    text: string;
    error: string | null;
  }>({ key: draftKey, text: initialText, error: null });
  const draft = draftState.key === draftKey ? draftState.text : initialText;
  const error = draftState.key === draftKey ? draftState.error : null;
  const updateDraft = (text: string) => {
    setDraftState({
      key: draftKey,
      text,
      error: draftState.key === draftKey ? draftState.error : null,
    });
  };
  const updateError = (nextError: string | null) => {
    setDraftState({ key: draftKey, text: draft, error: nextError });
  };
  const resetDraft = () => {
    setDraftState({ key: draftKey, text: initialText, error: null });
  };

  const apply = () => {
    if (!path) return;
    const parsed = parseSchemaEditorDraft(draft, editorMode);
    if (!parsed.ok) {
      updateError(parsed.error);
      return;
    }
    onApplyValue(path, parsed.value);
  };

  const hasIssue = path && validation
    ? [...validation.errors, ...validation.warnings].some((issue) =>
        issue.path ? normalizeGalleryPath(issue.path).startsWith(normalizeGalleryPath(path)) : false
      )
    : false;

  return (
    <div className="space-y-3">
      <div className="flex items-start justify-between gap-3">
        <div className="min-w-0">
          <div className="font-mono text-[10px] font-semibold uppercase tracking-wider text-muted-foreground">
            selected field
          </div>
          <div className="mt-1 truncate font-mono text-xs text-foreground">
            {path ? path.replace(/^\$\.?/, "") : "none"}
          </div>
          <div className="mt-1 truncate text-[11px] text-muted-foreground">
            current: {preview?.value ?? "parse pending"}
          </div>
        </div>
        <span
          className={cn(
            "shrink-0 rounded-md border px-2 py-1 font-mono text-[10px]",
            hasIssue ? "border-amber-500/50 text-amber-300" : "border-border text-muted-foreground"
          )}
        >
          {field?.type ?? "field"}
        </span>
      </div>

      {editorMode === "boolean" ? (
        <select
          value={draft}
          onChange={(event) => updateDraft(event.target.value)}
          className="h-9 w-full rounded-md border border-border bg-background px-2 font-mono text-xs text-foreground outline-none focus:border-brand-500/70"
        >
          <option value="true">true</option>
          <option value="false">false</option>
        </select>
      ) : editorMode === "enum" ? (
        <select
          value={draft}
          onChange={(event) => updateDraft(event.target.value)}
          className="h-9 w-full rounded-md border border-border bg-background px-2 font-mono text-xs text-foreground outline-none focus:border-brand-500/70"
        >
          {enumOptions.map((option) => (
            <option key={option} value={option}>
              {option}
            </option>
          ))}
        </select>
      ) : editorMode === "gallery" ? (
        <select
          value={draft}
          onChange={(event) => updateDraft(event.target.value)}
          className="h-9 w-full rounded-md border border-border bg-background px-2 font-mono text-xs text-foreground outline-none focus:border-brand-500/70"
        >
          {galleryItems.map((item) => (
            <option key={item.id} value={item.id}>
              {item.label}
            </option>
          ))}
        </select>
      ) : editorMode === "json" ? (
        <textarea
          value={draft}
          onChange={(event) => updateDraft(event.target.value)}
          className="min-h-24 w-full resize-y rounded-md border border-border bg-background px-2 py-2 font-mono text-[11px] text-foreground outline-none focus:border-brand-500/70"
        />
      ) : (
        <input
          type={editorMode === "number" ? "number" : "text"}
          value={draft}
          onChange={(event) => updateDraft(event.target.value)}
          className="h-9 w-full rounded-md border border-border bg-background px-2 font-mono text-xs text-foreground outline-none focus:border-brand-500/70"
        />
      )}

      {proposal ? (
        <button
          type="button"
          onClick={() => updateDraft(schemaEditorInitialText(undefined, proposal.value, editorMode))}
          className="rounded-md border border-border bg-background px-2 py-1 font-mono text-[11px] text-muted-foreground hover:bg-muted/40 hover:text-foreground"
        >
          insert schema default
        </button>
      ) : null}

      {error ? <p className="text-[11px] text-destructive">{error}</p> : null}

      <div className="flex items-center justify-between gap-2">
        <button
          type="button"
          onClick={resetDraft}
          className="inline-flex h-9 items-center rounded-lg border border-border bg-background px-3 font-mono text-xs font-semibold text-foreground hover:bg-muted/40 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
        >
          Reset
        </button>
        <button
          type="button"
          disabled={!document || !path || isSending}
          onClick={apply}
          className="inline-flex h-9 items-center gap-1.5 rounded-lg bg-brand-600 px-3 font-mono text-xs font-semibold text-white hover:bg-brand-700 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring disabled:cursor-not-allowed disabled:opacity-45"
        >
          <Check className="size-3.5" />
          Apply change
        </button>
      </div>
    </div>
  );
}

function manualToolRunnable(name: string) {
  return [
    "read.opf",
    "read.uploads",
    "read.gallery_sources",
    "resolve.field",
    "search.gallery",
    "validate.opf",
  ].includes(name);
}

function manualToolCanRun(name: string, document: OPFDocument | null) {
  if (!manualToolRunnable(name)) return false;
  if ((name === "read.opf" || name === "validate.opf" || name === "resolve.field") && !document) return false;
  return true;
}

function manualToolDisabledReason(tool: AuthorToolCatalogItem, document: OPFDocument | null) {
  if (!manualToolRunnable(tool.name)) {
    return `${tool.name} is reserved for the agent workflow in this pass.`;
  }
  if ((tool.name === "read.opf" || tool.name === "validate.opf") && !document) {
    return "Parse a valid OPF draft before running this tool.";
  }
  return tool.ui.summary;
}

function manualToolInput(
  name: string,
  input: {
    attachments: Attachment[];
    document: OPFDocument | null;
    gallerySources: ReadonlyArray<AuthorGallerySource>;
    kind: GalleryKind | "all";
    query: string;
    resolveLayers: "all" | "winning-only";
    resolvePath: string;
    selectedPath: string | null;
  }
) {
  if (name === "read.opf") {
    return { document: input.document ?? undefined, path: input.selectedPath ?? undefined };
  }
  if (name === "read.uploads") {
    return { uploads: input.attachments };
  }
  if (name === "read.gallery_sources") {
    return { sources: input.gallerySources };
  }
  if (name === "validate.opf") {
    return { document: input.document };
  }
  if (name === "resolve.field") {
    return {
      document: input.document ?? undefined,
      gallerySources: input.gallerySources,
      layers: input.resolveLayers,
      path: input.resolvePath.trim() || input.selectedPath || "narrative",
    };
  }
  return {
    document: input.document ?? undefined,
    query: input.query,
    kind: input.kind === "all" ? undefined : input.kind,
    limit: 8,
    sources: input.gallerySources,
  };
}

function manualToolCallSummary(name: string, result: unknown) {
  if (name === "resolve.field") {
    const winningLayer = isRecord(result) && isRecord(result.result) ? result.result.winningLayer : undefined;
    return { outputSummary: `Resolved field${winningLayer ? ` · layer ${winningLayer}` : ""}.` };
  }
  if (name === "search.gallery") {
    const items = isRecord(result) && isRecord(result.result) && Array.isArray(result.result.items)
      ? result.result.items
      : [];
    return { outputSummary: `Found ${items.length} gallery item${items.length === 1 ? "" : "s"}.` };
  }
  if (name === "validate.opf") {
    const valid = isRecord(result) && isRecord(result.result) ? result.result.valid : undefined;
    return { outputSummary: valid ? "OPF is valid." : "Validation returned issues." };
  }
  if (name === "read.uploads") {
    const count = isRecord(result) && isRecord(result.result) ? result.result.count : 0;
    return { outputSummary: `Read ${count} upload${count === 1 ? "" : "s"}.` };
  }
  if (name === "read.gallery_sources") {
    const count = isRecord(result) && isRecord(result.result) ? result.result.count : 0;
    return { outputSummary: `Read ${count} gallery source${count === 1 ? "" : "s"}.` };
  }
  const path = isRecord(result) && isRecord(result.result) && typeof result.result.path === "string"
    ? result.result.path
    : undefined;
  return {
    outputSummary: `Read ${path ?? "OPF"}.`,
    affectedPaths: path ? [path] : undefined,
  };
}

type SchemaEditorMode = "boolean" | "enum" | "gallery" | "json" | "number" | "string";

function schemaEditorMode({
  currentValue,
  enumOptions,
  fieldType,
  galleryContext,
}: {
  currentValue: unknown;
  enumOptions: string[];
  fieldType?: string;
  galleryContext: GalleryContext | null;
}): SchemaEditorMode {
  if (galleryContext) return "gallery";
  if (enumOptions.length > 0) return "enum";
  if (typeof currentValue === "boolean" || fieldType === "boolean") return "boolean";
  if (typeof currentValue === "number" || /\b(number|integer)\b/.test(fieldType ?? "")) return "number";
  if (Array.isArray(currentValue) || (currentValue && typeof currentValue === "object")) return "json";
  if (/\b(array|object|Record|\[\])\b/i.test(fieldType ?? "")) return "json";
  return "string";
}

function schemaEditorInitialText(
  currentValue: unknown,
  proposalValue: unknown,
  mode: SchemaEditorMode
) {
  const value = currentValue === undefined ? proposalValue : currentValue;
  if (mode === "gallery") return galleryIdFromValue(value) ?? "";
  if (mode === "json") return JSON.stringify(value ?? null, null, 2);
  if (mode === "boolean") return value === true ? "true" : "false";
  if (value === undefined || value === null) return "";
  return String(value);
}

function parseSchemaEditorDraft(
  draft: string,
  mode: SchemaEditorMode
): { ok: true; value: unknown } | { ok: false; error: string } {
  if (mode === "json") {
    try {
      return { ok: true, value: JSON.parse(draft) };
    } catch (error) {
      return { ok: false, error: error instanceof Error ? error.message : "Invalid JSON." };
    }
  }
  if (mode === "number") {
    const value = Number(draft);
    if (!Number.isFinite(value)) return { ok: false, error: "Enter a valid number." };
    return { ok: true, value };
  }
  if (mode === "boolean") return { ok: true, value: draft === "true" };
  return { ok: true, value: draft };
}

function schemaChoiceOptions(path: string | null, fieldType?: string) {
  const normalized = path ? normalizeGalleryPath(path) : "";
  if (normalized.endsWith(".role") || normalized === "organization.role") {
    return ["primary", "partner", "client", "sponsor", "host"];
  }
  if (normalized === "language") return ["en-US", "en", "es", "fr", "de"];
  if (fieldType === "enum") return [];
  return [];
}

function TracePane({
  centerTab,
  copiedPayload,
  document,

  onTogglePin,

  onSelectRun: (runId: string) => void;
  onOpenLibrary: (kind?: GalleryKind | "all" | "designs") => void;
  onUseGalleryCandidate: (path: string, item: GalleryItem) => void | Promise<void>;
  payloadPreview: { ok: true; payload: ApiPayloadPreview } | { ok: false; error: string };
  rejectedRunCount: number;
  runs: PublicAuthorRun[];
  selectedRun: PublicAuthorRun | null;
  surfaceError: string | null;
  issues: ReadonlyArray<ValidationIssue>;
  onJumpToPath: (path: string) => void;
  previewDocument: OPFDocument | null;
  selectedPreviewSlideIndex: number;
  selectionPath: string | null;
  selectionRows: ReadonlyArray<SelectionRow>;
  selectionTypeSignature: string | undefined;
  selectionPinned: boolean;
  onTogglePin: () => void;

  const [activeTab, setActiveTab] = useState<RightRailTab>(() => readRightRailTab());
  const [gallerySourceFilter, setGallerySourceFilter] = useState("all");

  const galleryContext = useMemo(
    () => galleryContextForPath(document, selectionPath, "selection"),
    [document, selectionPath]
  );
  const galleryContextKeyValue = galleryContext ? galleryContextKey(galleryContext) : "empty";
  const galleryUsageMap = useMemo(() => galleryUsage(document), [document]);
  const galleryAllItems = useMemo(
    () => (galleryContext ? galleryItemsForContext(galleryContext, document, "", gallerySources) : []),
    [document, galleryContext, gallerySources]
  );
  const gallerySourceFilters = useMemo<GalleryRailSourceFilter[]>(() => {
    const counts = new Map<string, number>();
    for (const item of galleryAllItems) {
      counts.set(item.sourceLabel, (counts.get(item.sourceLabel) ?? 0) + 1);
    }
    return [
      { id: "all", label: "all", count: galleryAllItems.length },
      ...[...counts.entries()].map(([label, count]) => ({ id: label, label, count })),
    ];
  }, [galleryAllItems]);
  const galleryFilteredItems = useMemo(
    () =>
      gallerySourceFilter === "all"
        ? galleryAllItems
        : galleryAllItems.filter((item) => item.sourceLabel === gallerySourceFilter),
    [galleryAllItems, gallerySourceFilter]
  );
  const galleryCandidates = useMemo<GalleryRailCandidate[]>(() => {
    if (!galleryContext) return [];
    return galleryFilteredItems.map((item) => {
      const usage = galleryUsageMap.get(galleryUsageKey(item.kind, item.id));
      return {
        id: galleryCandidateKey(item),
        label: item.label,
        subtitle: item.subtitle,
        sourceLabel: item.sourceLabel,
        swatches: item.swatches,
        summary: item.summary,
        usageCount: usage?.count ?? 0,
        inUse: galleryContext.currentId === item.id || Boolean(usage?.count),
        raw: item.raw ?? item.value ?? item,
      };
    });
  }, [galleryContext, galleryFilteredItems, galleryUsageMap]);
  const galleryRailContext: GalleryRailContext | null = galleryContext
    ? {
        kind: galleryContext.kind,
        title: galleryContext.title,
        path: galleryContext.path,
        scope: galleryContext.scope,
        currentId: galleryContext.currentId,
      }
    : null;
  const resolutionResult = useMemo(
    () =>
      selectionPath
        ? resolveField({
            document,
            gallerySources,
            path: selectionPath,
          })
        : null,
    [document, gallerySources, selectionPath]
  );
  const resolutionUsage = useMemo(
    () => buildResolutionUsage(document, selectionPath),
    [document, selectionPath]
  );
  const hasResolutionStack =
    Boolean(resolutionResult && resolutionResult.errors.length === 0 && resolutionResult.winningLayer != null);

  useEffect(() => {
    if (typeof window === "undefined") return;
    window.localStorage.setItem(RIGHT_RAIL_TAB_STORAGE_KEY, activeTab);
  }, [activeTab]);

  useEffect(() => {
    // eslint-disable-next-line react-hooks/set-state-in-effect
    setGallerySourceFilter("all");
  }, [galleryContextKeyValue]);

  const runTimeline = (
    <RunTimeline
      onSelectRun={onSelectRun}
      rejectedRunCount={rejectedRunCount}
      runs={runs}
      selectedRun={selectedRun}
    />
  );
  const handleUseGalleryCandidate = (candidateId: string) => {
    if (!galleryContext?.path) return;
    const item = galleryAllItems.find((candidate) => galleryCandidateKey(candidate) === candidateId);
    if (!item) return;
    void onUseGalleryCandidate(galleryContext.path, item);
  };

  return (
    <div className="min-h-0 flex-1 overflow-y-auto">
      {surfaceError ? (
        <div className="m-4 rounded-xl border border-destructive/40 bg-destructive/10 px-3 py-3 text-sm text-destructive">
          {surfaceError}
        </div>
      ) : null}

      <RightRailTabs activeTab={activeTab} onChange={setActiveTab} />

      {activeTab === "inspector" ? (
        centerTab === "preview" ? (
          <SlideInspector
            document={previewDocument}
            selectedSlideIndex={selectedPreviewSlideIndex}
            onJumpToSlide={onPreviewSlideSelect}
            onSelectSlide={onPreviewSlideSelect}
            onSwapLayout={(index) => {
              onPreviewPathSelect(`$.slides[${index}].layout`);
              setActiveTab("gallery");
            }}
            onSwapTheme={() => {
              onPreviewPathSelect("$.design.theme");
              setActiveTab("gallery");
            }}
          />
        ) : (
            hasResolutionStack && resolutionResult && selectionPath ? (
              <ResolutionInspector
                onJumpToPath={onJumpToPath}
                onTogglePin={onTogglePin}
                path={selectionPath}
                pinned={selectionPinned}
                result={resolutionResult}
                typeSignature={selectionTypeSignature}
                usage={resolutionUsage}
                validationIssues={issues}
              />
            ) : (
              <Inspector
                issues={issues}
                onJumpToPath={onJumpToPath}
                selectionPath={selectionPath}
                selectionRows={selectionRows}
                selectionTypeSignature={selectionTypeSignature}
                selectionPinned={selectionPinned}
                onTogglePin={onTogglePin}
              />
            )
        )
      ) : activeTab === "gallery" ? (
        <GalleryRailTab
          candidates={galleryCandidates}
          context={galleryRailContext}
            onBrowseAll={() => onOpenLibrary(galleryContext?.kind ?? "all")}
          onSourceFilterChange={setGallerySourceFilter}
          onUseCandidate={handleUseGalleryCandidate}
          sourceFilter={gallerySourceFilter}
          sourceFilters={gallerySourceFilters}
        />
      ) : (
        <Trace
          timeline={runTimeline}
          runDetail={
            selectedRun ? <RunDetailCard run={selectedRun} onRetry={onRetryRun} /> : null
          }
          apiPayload={
            <ApiPayloadCard
              copied={copiedPayload}
              onCopy={onCopyPayload}
              payloadPreview={payloadPreview}
            />
          }
        />
      )}
    </div>
  );
}

function buildResolutionUsage(
  document: OPFDocument | null,
  path: string | null
): ResolutionUsageChip[] {
  if (!document || !path) return [];
  const normalized = path.replace(/^\$\.?/, "");
  const slideLayoutMatch = /^slides\[(\d+)]\.layout$/.exec(normalized);
  if (slideLayoutMatch) {
    const index = Number(slideLayoutMatch[1]);
    const slide = document.slides?.[index];
    return [
      {
        label: `slides[${index}]`,
        path: `$.slides[${index}]`,
        description:
          typeof slide?.title === "string"
            ? slide.title
            : typeof slide?.id === "string"
              ? slide.id
              : "Selected slide",
      },
    ];
  }

  if (isDeckLevelResolutionPath(normalized)) {
    return (document.slides ?? []).slice(0, 8).map((slide, index) => ({
      label: `slides[${index}]`,
      path: `$.slides[${index}]`,
      description:
        typeof slide.title === "string"
          ? slide.title
          : typeof slide.id === "string"
            ? slide.id
            : `Slide ${index + 1}`,
    }));
  }

  return [{ label: normalized, path }];
}

function isDeckLevelResolutionPath(path: string): boolean {
  return (
    path === "narrative" ||
    path === "tone" ||
    path === "audience" ||
    path.startsWith("audience") ||
    path === "design.theme" ||
    path === "design.colorScheme" ||
    path === "design.fontScheme"
  );
}

function buildPayloadPreview({
  attachments,
  baseOpfDocumentId,
  inputMode,
  operationMode,
  pinnedPaths,
  prompt,
  rawText,
  remoteAuthorAvailable,
  selectionPath,
  threadId,
}: {
  attachments: Attachment[];
  baseOpfDocumentId?: string;
  inputMode: InputMode;
  operationMode: OperationMode;
  pinnedPaths?: ReadonlyArray<string>;
  prompt: string;
  rawText: string;
  remoteAuthorAvailable: boolean;
  selectionPath?: string | null;
  threadId?: string;
}): { ok: true; payload: ApiPayloadPreview } | { ok: false; error: string } {
  if (!threadId) {
    if (!remoteAuthorAvailable) {
      return {
        ok: false,
        error: "Sign in to create saved Author threads or use AI operations.",
      };
    }
    if (inputMode !== "raw") {
      const prepared = prepareAgentMessageText({ attachments, inputMode, prompt });
      if (!prepared.ok) return prepared;
      return {
        ok: true,
        payload: {
          method: "POST",
          endpoint: "/api/v1/author/threads",
          body: {
            client: { surface: "author-playground", chat: "agent-loop" },
          },
        },
      };
    }
    const prepared = prepareCompileInput({ attachments, inputMode, prompt, rawText });
    if (!prepared.ok) return prepared;
    return {
      ok: true,
      payload: {
        method: "POST",
        endpoint: "/api/v1/author/threads",
        body: {
          input: prepared.request,
          client: { surface: "author-playground" },
        },
      },
    };
  }

  if (operationMode === "create") {
    if (inputMode !== "raw") {
      const prepared = prepareAgentMessageText({ attachments, inputMode, prompt });
      if (!prepared.ok) return prepared;
      return {
        ok: true,
        payload: {
          method: "POST",
          endpoint: `/api/v1/author/threads/${threadId}/messages`,
          body: {
            operation: "create",
            message: prepared.text,
          },
        },
      };
    }
    const prepared = prepareCompileInput({ attachments, inputMode, prompt, rawText });
    if (!prepared.ok) return prepared;
    return {
      ok: true,
      payload: {
        method: "POST",
        endpoint: `/api/v1/author/threads/${threadId}/messages`,
        body: {
          operation: "create",
          message: contentFromCompileRequest(prepared.request),
          input: prepared.request,
        },
      },
    };
  }

  const message = prompt.trim();
  if (!message) return { ok: false, error: "Write a message before sending." };
  if (message.length > MAX_INPUT_LENGTH) return { ok: false, error: "Message exceeds 50KB." };

  const body: Record<string, unknown> = {
    message,
  };
  if (operationMode !== "auto") body.operation = operationMode;
  if (baseOpfDocumentId) body.baseOpfDocumentId = baseOpfDocumentId;
  const context: Record<string, unknown> = {};
  if (selectionPath) context.selectionPath = selectionPath;
  const cleanedPinnedPaths = (pinnedPaths ?? [])
    .map((path) => path.trim())
    .filter(Boolean)
    .slice(0, 10);
  if (cleanedPinnedPaths.length > 0) context.pinnedPaths = cleanedPinnedPaths;
  if (Object.keys(context).length > 0) body.context = context;

  return {
    ok: true,
    payload: {
      method: "POST",
      endpoint: `/api/v1/author/threads/${threadId}/messages`,
      body,
    },
  };
}

function prepareAgentMessageText({
  attachments,
  inputMode,
  prompt,
}: {
  attachments: Attachment[];
  inputMode: InputMode;
  prompt: string;
}): { ok: true; text: string } | { ok: false; error: string } {
  const trimmedPrompt = prompt.trim();
  if (trimmedPrompt.length > MAX_INPUT_LENGTH) {
    return { ok: false, error: "Prompt exceeds 50KB." };
  }
  if (!trimmedPrompt && attachments.length === 0) {
    return { ok: false, error: "Write a message before sending." };
  }

  const uploads = authorUploads(attachments);
  if (uploads.length === 0) return { ok: true, text: trimmedPrompt };

  const uploadText = uploads
    .map((upload, index) => {
      const header = `Source ${index + 1}: ${upload.name} (${upload.type || "file"}, ${formatBytes(upload.size)})`;
      return upload.text?.trim()
        ? `${header}\n${upload.text.trim()}`
        : `${header}\nNo text could be extracted from this file.`;
    })
    .join("\n\n");
  const instruction =
    inputMode === "upload" && !trimmedPrompt
      ? "Create a presentation from the attached source files."
      : trimmedPrompt;
  return {
    ok: true,
    text: `${instruction}\n\nAttached source material:\n${uploadText}`,
  };
}

function authorAgentRequestContext({
  pinnedPaths,
  selectionPath,
}: {
  pinnedPaths: ReadonlyArray<string>;
  selectionPath?: string | null;
}): { pinnedPaths: string[]; selectionPath?: string } {
  const context: { pinnedPaths: string[]; selectionPath?: string } = {
    pinnedPaths: pinnedPaths.map((path) => path.trim()).filter(Boolean).slice(0, 10),
  };
  if (selectionPath) context.selectionPath = selectionPath;
  return context;
}

function prepareCompileInput({
  attachments,
  inputMode,
  prompt,
  rawText,
}: {
  attachments: Attachment[];
  inputMode: InputMode;
  prompt: string;
  rawText: string;
}): { ok: true; request: AuthorCompileRequest } | { ok: false; error: string } {
  if (inputMode === "raw") {
    const text = rawText.trim();
    if (!text) return { ok: false, error: "Paste raw source text or OPF JSON before creating a draft." };
    if (text.length > MAX_INPUT_LENGTH) return { ok: false, error: "Raw input exceeds 50KB." };
    return { ok: true, request: { mode: "raw", rawText: text, uploads: authorUploads(attachments) } };
  }

  const trimmedPrompt = prompt.trim();
  if (trimmedPrompt.length > MAX_INPUT_LENGTH) return { ok: false, error: "Prompt exceeds 50KB." };
  if (!trimmedPrompt && attachments.length === 0) {
    return { ok: false, error: "Add a prompt or upload a source file first." };
  }
  return {
    ok: true,
    request: {
      mode: inputMode,
      prompt: trimmedPrompt,
      uploads: authorUploads(attachments),
    },
  };
}

function galleryContextForPath(
  document: OPFDocument | null,
  path: string | null,
  source: GalleryContext["source"]
): GalleryContext | null {
  if (!path) return null;
  const normalized = normalizeGalleryPath(path);
  if (!normalized) return null;

  const slideLayoutMatch = /^slides\[(\d+)]\.layout$/.exec(normalized);
  if (slideLayoutMatch) {
    const canonicalPath = `$.${normalized}`;
    const current = document ? readValueAtPath(document, canonicalPath) : undefined;
    const slideIndex = Number(slideLayoutMatch[1]);
    return {
      kind: "layouts",
      path: canonicalPath,
      title: "layouts",
      scope: layoutScopeForSlide(document, slideIndex),
      currentId: galleryIdFromValue(current),
      currentLabel: galleryCurrentLabel(current),
      source,
    };
  }

  if (normalized === "design.colorScheme" || normalized.startsWith("design.colorScheme.")) {
    const current = document ? readValueAtPath(document, "$.design.colorScheme") : undefined;
    return {
      kind: "colorSchemes",
      path: "$.design.colorScheme",
      title: "colors",
      scope: "all schemes",
      currentId: galleryIdFromValue(current),
      currentLabel: galleryCurrentLabel(current),
      source,
    };
  }

  if (normalized === "design.fontScheme" || normalized.startsWith("design.fontScheme.")) {
    const current = document ? readValueAtPath(document, "$.design.fontScheme") : undefined;
    return {
      kind: "fontSchemes",
      path: "$.design.fontScheme",
      title: "fonts",
      scope: "all pairings",
      currentId: galleryIdFromValue(current),
      currentLabel: galleryCurrentLabel(current),
      source,
    };
  }

  if (normalized === "design.theme" || normalized.startsWith("design.theme.")) {
    const current = document ? readValueAtPath(document, "$.design.theme") : undefined;
    return {
      kind: "themes",
      path: "$.design.theme",
      title: "themes",
      scope: "deck themes",

function manualGalleryTarget(
  kind: GalleryKind,
  document: OPFDocument | null,
  selectedPath: string | null,
  sourceContext: GalleryContext | null
): Pick<GalleryContext, "currentId" | "currentLabel" | "path" | "scope"> | null {
  if (sourceContext?.kind === kind && sourceContext.path) {
    return {
      currentId: sourceContext.currentId,
      currentLabel: sourceContext.currentLabel,
      path: sourceContext.path,
      scope: sourceContext.scope,
    };
  }

  const slideIndex = selectedSlideIndexForGallery(document, selectedPath);
  const path =
    kind === "layouts" && slideIndex != null
      ? `$.slides[${slideIndex}].layout`
      : kind === "colorSchemes"
          ? "$.design.colorScheme"
          : kind === "fontSchemes"
            ? "$.design.fontScheme"
            : kind === "themes"
              ? "$.design.theme"
            : kind === "tones"
              ? "$.tone"
              : kind === "narratives"
                ? "$.narrative"
                : kind === "audiences"
                  ? "$.audience"
                  : kind === "languages"
                    ? "$.language"
                : null;
  if (!path) return null;
  const current = document ? readValueAtPath(document, path) : undefined;
  return {
    currentId: galleryIdFromValue(current),
    currentLabel: galleryCurrentLabel(current),
    path,
    scope:
      kind === "layouts" && slideIndex != null
        ? layoutScopeForSlide(document, slideIndex)
        : `all ${GALLERY_META[kind].label}`,
  };
}

function galleryItemsForContext(
  context: GalleryContext,
  document: OPFDocument | null,
  search: string,
  sources: ReadonlyArray<AuthorGallerySource> = []
): GalleryItem[] {
  const allItems = galleryItemsForKind(context.kind, document, sources);
  let items = allItems;
  if (context.kind === "layouts" && context.scope.startsWith("bullets")) {
    items = allItems.filter(
      (item) =>
        item.featured ||
        item.subtitle.includes("text") ||
        item.id.includes("text") ||
        item.id.includes("bullet") ||
        item.id.includes("list")
    );
  }
  if (context.currentId) {
    items = ensureGalleryItem(items, context.kind, context.currentId);
  }
  return filterGalleryItems(items, search);
}

function galleryItemsForKind(
  kind: GalleryKind,

function filterGalleryItems(items: GalleryItem[], search: string): GalleryItem[] {
  const query = search.trim().toLowerCase();
  if (!query) return items;
  return items.filter((item) =>
    [item.id, item.label, item.subtitle, item.summary]
      .filter(Boolean)
      .some((value) => String(value).toLowerCase().includes(query))
  );
}

function galleryUsage(document: OPFDocument | null): GalleryUsage {
  const usage: GalleryUsage = new Map();
  if (!document) return usage;

  const record = (kind: GalleryKind, value: unknown, path: string) => {
    const id = galleryIdFromValue(value);
    if (!id) return;
    const key = galleryUsageKey(kind, id);
    const entry = usage.get(key) ?? { count: 0, paths: [] };
    entry.count += 1;
    entry.paths.push(path);
    usage.set(key, entry);
  };

  record("themes", readValueAtPath(document, "$.design.theme"), "$.design.theme");
  record("colorSchemes", readValueAtPath(document, "$.design.colorScheme"), "$.design.colorScheme");
  record("fontSchemes", readValueAtPath(document, "$.design.fontScheme"), "$.design.fontScheme");
  record("tones", readValueAtPath(document, "$.tone"), "$.tone");
  record("narratives", readValueAtPath(document, "$.narrative"), "$.narrative");
  record("audiences", readValueAtPath(document, "$.audience"), "$.audience");
  record("languages", readValueAtPath(document, "$.language"), "$.language");

  document.slides?.forEach((slide, index) => {
    record("layouts", slide.layout, `$.slides[${index}].layout`);
    if (isRecord(slide.chart)) {
      record("chartTypes", slide.chart.type, `$.slides[${index}].chart.type`);
    }
  });

  return usage;
}

function galleryDocumentRefs(
  document: OPFDocument | null,
  usage: GalleryUsage,
  sources: ReadonlyArray<AuthorGallerySource> = []
): GalleryDocumentRef[] {
  if (!document) return [];
  return [...usage.entries()]
    .map(([key, entry]) => {
      const parsed = parseGalleryUsageKey(key);
      if (!parsed) return null;
      const item = galleryItemForId(parsed.kind, parsed.id, document, sources) ?? fallbackGalleryItem(parsed.kind, parsed.id);
      return {
        kind: parsed.kind,
        id: parsed.id,
        count: entry.count,
        path: entry.paths[0] ?? "$",
        item,
      };
    })
    .filter((item): item is GalleryDocumentRef => Boolean(item))
    .sort((a, b) => GALLERY_KINDS.indexOf(a.kind) - GALLERY_KINDS.indexOf(b.kind));
}

function setValueAtOpfPath(document: OPFDocument, path: string, value: unknown): boolean {
  const segments = opfPathSegments(path);
  if (!segments || segments.length === 0) return false;
  let current: unknown = document;
  for (let index = 0; index < segments.length - 1; index += 1) {
    const segment = segments[index];
    const nextSegment = segments[index + 1];
    if (typeof segment === "number") {
      if (!Array.isArray(current) || segment < 0 || segment >= current.length) return false;
      current = current[segment];
      continue;
    }
    if (!isRecord(current)) return false;
    if (!isRecord(current[segment]) && !Array.isArray(current[segment])) {
      current[segment] = typeof nextSegment === "number" ? [] : {};
    }
    current = current[segment];
  }

  const leaf = segments[segments.length - 1];
  if (typeof leaf === "number") {
    if (!Array.isArray(current) || leaf < 0 || leaf > current.length) return false;
    current[leaf] = value;
    return true;
  }
  if (!isRecord(current)) return false;
  current[leaf] = value;
  return true;
}

function opfPathSegments(path: string): Array<string | number> | null {
  const body = normalizeGalleryPath(path);
  if (!body) return [];
  const segments: Array<string | number> = [];
  let index = 0;
  while (index < body.length) {
    if (body[index] === ".") {
      index += 1;
      continue;
    }
    if (body[index] === "[") {
      const close = body.indexOf("]", index);
      if (close < 0) return null;
      const rawIndex = body.slice(index + 1, close);
      if (!/^\d+$/.test(rawIndex)) return null;
      segments.push(Number(rawIndex));
      index = close + 1;
      continue;
    }
    let end = index;
    while (end < body.length && body[end] !== "." && body[end] !== "[") end += 1;
    const key = body.slice(index, end);
    if (!key) return null;
    segments.push(key);
    index = end;
  }
  return segments;
}

function galleryItemForId(
  kind: GalleryKind,
  id: string,
  document: OPFDocument | null = null,

function ensureGalleryItem(
  items: GalleryItem[],
  kind: GalleryKind,
  id: string
): GalleryItem[] {
  if (items.some((item) => item.id === id)) return items;
  return [galleryItemForId(kind, id) ?? fallbackGalleryItem(kind, id), ...items];
}

  return {
    kind,
    id,
    label: id,
    subtitle: "authored",

function galleryItemUseCount(usage: GalleryUsage, kind: GalleryKind, id: string): number {
  return usage.get(galleryUsageKey(kind, id))?.count ?? 0;
}

function galleryKindUseCount(usage: GalleryUsage, kind: GalleryKind): number {
  let count = 0;
  for (const key of usage.keys()) {
    const parsed = parseGalleryUsageKey(key);
    if (parsed?.kind === kind) count += usage.get(key)?.count ?? 0;
  }
  return count;
}

function galleryUsageKey(kind: GalleryKind, id: string): string {
  return `${kind}:${id}`;
}

function parseGalleryUsageKey(key: string): { kind: GalleryKind; id: string } | null {
  const separator = key.indexOf(":");
  if (separator < 0) return null;
  const kind = key.slice(0, separator) as GalleryKind;
  if (!GALLERY_KINDS.includes(kind)) return null;
  return { kind, id: key.slice(separator + 1) };
}

function layoutScopeForSlide(document: OPFDocument | null, slideIndex: number): string {
  const slide = document?.slides?.[slideIndex];
  if (!slide || !isRecord(slide)) return "all layouts";
  if (Array.isArray(slide.bullets) || Array.isArray(slide.items)) return "bullets variants";
  if (slide.image) return "image variants";
  if (slide.chart) return "chart variants";
  if (slide.table) return "table variants";
  if (slide.quote) return "quote variants";
  return "all layouts";
}

function selectedSlideIndexForGallery(document: OPFDocument | null, path: string | null): number | null {
  const count = document?.slides?.length ?? 0;
  if (count === 0) return null;
  const normalized = path ? normalizeGalleryPath(path) : "";
  const match = /^slides\[(\d+)]/.exec(normalized);
  if (match) {
    const index = Number(match[1]);
    return index >= 0 && index < count ? index : null;
  }
  return count > 2 ? 2 : 0;
}

function normalizeGalleryPath(path: string): string {
  const trimmed = path.trim();
  if (trimmed === '["$schema"]' || trimmed === '$["$schema"]') return "$schema";
  return trimmed.replace(/^\$\.?/, "");
}

function galleryIdFromValue(value: unknown): string | null {
  if (typeof value === "string" && value.trim()) return value.trim();
  if (Array.isArray(value)) return galleryIdFromValue(value[0]);
  if (!isRecord(value)) return null;
  for (const key of ["id", "slug", "scheme", "template", "type"]) {
    const candidate = value[key];
    if (typeof candidate === "string" && candidate.trim()) return candidate.trim();
  }
  return null;
}

function galleryCurrentLabel(value: unknown): string {
  const id = galleryIdFromValue(value);
  if (id) return id;
  return formatResolvedValue(value);
}

function parseDocument(opfText: string, format: Format = "json") {
  if (!opfText.trim()) return { document: null as OPFDocument | null, error: null as string | null };
  const result = parseText(opfText, format);
  if (!result.ok) return { document: null as OPFDocument | null, error: result.error };
  if (!result.doc || typeof result.doc !== "object") {
    return { document: null as OPFDocument | null, error: "Document must be an object." };
  }
  return { document: result.doc as OPFDocument, error: null };
}

function buildHandoffOpfDocument(doc: unknown): PublicAuthorOpfDocument {
  if (!doc || typeof doc !== "object" || Array.isArray(doc)) {
    throw new Error("Handoff payload must be an OPF JSON object.");
  }

  const opf = doc as OPFDocument;
  const validation = validateOPF(opf);
  return {
    id: "handoff-opf",
    threadId: "handoff",
    version: 1,
    opf,
    validation,
    createdAt: new Date(0).toISOString(),
  };
}

};

function summarizeDeck(document: OPFDocument | null): DeckSummary {
  if (!document) {
    return { audience: null, durationMinutes: null, filenameBase: null, slideCount: 0, tone: null };
  }
  const narrative: Record<string, unknown> = isRecord(document.narrative) ? document.narrative : {};
  const title = typeof document.name === "string" ? document.name : "";
  return {
    audience: firstString(document.audience),
    durationMinutes: firstNumber(narrative.durationMinutes) ?? firstNumber(document.duration),
    filenameBase: resolveOpfFilenameBase(document, title),
    slideCount: countSlides(document),
    tone: firstString(narrative.tone) ?? firstString(document.tone),
  };
}

function resolveOpfFilenameBase(document: OPFDocument, title: string): string {
  const resolved = resolveOpfFilename(
    { filename: firstString(document.filename), title },
    "pptx"
  );
  return stripAuthorSourceExtension(resolved.replace(/\.pptx$/i, "")) || "presentation";
}

  const result = validateOpfFilename(input, "filename");
  if (!result.ok) return result;
  return {
    ...result,
    value: stripAuthorSourceExtension(result.value),
  };
}

function stripAuthorSourceExtension(input: string) {
  return input.replace(/(\.opf)?\.(json|yaml|yml|md|markdown)$/i, "");
}

function opfTextForFormat(document: OPFDocument, format: Format) {
  const jsonText = toJsonText(document);
  if (format === "json") return jsonText;
  const converted = convert(jsonText, "json", format);

function defaultAuthorSelectionPath(document: OPFDocument | null): string | null {
  if (!document || !Array.isArray(document.slides) || document.slides.length === 0) return null;
  const slide = document.slides[2] ?? document.slides[0];
  const index = document.slides[2] ? 2 : 0;
  if (isRecord(slide)) {
    if (isRecord(slide.right) && Array.isArray(slide.right.items)) {
      return `$.slides[${index}].right.items`;
    }
    if (Array.isArray(slide.items)) return `$.slides[${index}].items`;
    if (typeof slide.title === "string") return `$.slides[${index}].title`;
  }
  return "$.slides";
}

function firstString(value: unknown) {
  if (typeof value === "string" && value.trim()) return value.trim();
  if (Array.isArray(value)) {
    const first = value.find((item) => typeof item === "string" && item.trim());
    return typeof first === "string" ? first.trim() : null;
  }
  return null;
}

function firstNumber(value: unknown) {
  return typeof value === "number" && Number.isFinite(value) ? value : null;
}

  } catch {
    return iso;
  }
}

/**
 * Map a cursor schema path to a small set of suggested prompts.
 *
 * Order matters: more-specific path patterns first. Returns at most 4 chips.
 * Returns [] for paths we don't yet know how to suggest for — the composer
 * hides the chip row entirely in that case.
 */
function chipsForPath(path: string | null): ChipSuggestion[] {
  if (!path) return [];
  const trimmed = path.replace(/^\$\.?/, "");
  if (trimmed === "" || trimmed === "$") return [];

  // slides[].bullets
  if (/^slides\[\d+\]\.bullets/.test(trimmed)) {
    return [
      { id: "add-bullet", label: "+ add bullet", prompt: "Add a bullet to this slide." },
      { id: "reorder", label: "reorder", prompt: "Reorder these bullets for impact." },
      { id: "make-concise", label: "make concise", prompt: "Tighten these bullets — fewer words, same meaning." },
    ];
  }

  // slides[].title
  if (/^slides\[\d+\]\.title$/.test(trimmed)) {
    return [
      { id: "shorten", label: "shorten", prompt: "Shorten this slide title." },
      { id: "rephrase", label: "rephrase", prompt: "Rephrase this slide title." },
      { id: "add-subtitle", label: "+ add subtitle", prompt: "Add a subtitle to this slide." },
    ];
  }

  // slides[].notes
  if (/^slides\[\d+\]\.notes$/.test(trimmed)) {
    return [
      { id: "expand-notes", label: "expand", prompt: "Expand these speaker notes with more detail." },
      { id: "trim-notes", label: "trim", prompt: "Trim these speaker notes to the essentials." },
    ];
  }

  // top-level slides
  if (trimmed === "slides" || /^slides$/.test(trimmed)) {
    return [
      { id: "add-slide", label: "+ add slide", prompt: "Add a new slide." },
      { id: "rebalance", label: "rebalance", prompt: "Rebalance the deck — split or merge slides as needed." },
    ];
  }

  // design
  if (trimmed.startsWith("design")) {
    return [
      { id: "warmer", label: "warmer", prompt: "Adjust the design to feel warmer and more inviting." },
      { id: "tighter", label: "tighter", prompt: "Tighten the design — less ornamentation." },
    ];
  }

  // narrative / tone
  if (trimmed.startsWith("narrative") || trimmed.startsWith("tone")) {
    return [
      { id: "more-persuasive", label: "more persuasive", prompt: "Make this more persuasive without losing accuracy." },
      { id: "more-direct", label: "more direct", prompt: "Make this more direct — cut the hedging." },
    ];
  }

  // generic
  return [
    { id: "rewrite-field", label: "rewrite this field", prompt: `Rewrite ${trimmed}.` },
  ];
}

    return true;
  });
}

function formatBytes(size: number) {
  if (size < 1024) return `${size} B`;
  if (size < 1024 * 1024) return `${Math.round(size / 1024)} KB`;
  return `${(size / (1024 * 1024)).toFixed(1)} MB`;
}

function acceptedHunkIdsForGhostEdit(edit: GhostEdit) {
  return edit.hunks
    .filter((hunk) => edit.decisions?.[hunk.id] === "accept")
    .map((hunk) => hunk.id);
}

function statusLabel(status: AuthorRunStatus) {
  if (status === "generated") return "generated";
  if (status === "applied") return "applied";
  if (status === "rejected") return "rejected";
  if (status === "needs_input") return "needs input";
  if (status === "needs_review") return "needs review";
  return "error";
}

function shortId(id: string) {
  if (id.length <= 16) return id;
  return `${id.slice(0, 10)}...${id.slice(-4)}`;
}

function formatDateTime(value: string) {
  const date = new Date(value);
  if (Number.isNaN(date.getTime())) return value;
  return date.toLocaleTimeString([], { hour: "2-digit", minute: "2-digit" });
}

function formatPatchValue(value: unknown) {
  if (typeof value === "string") return JSON.stringify(value);
  return JSON.stringify(value);
}

type ApiErrorBody = {
  error?: {
    message?: string;
  };
  message?: string;
};

function isAuthorThreadSnapshot(value: unknown): value is AuthorThreadSnapshot {
  return Boolean(
    value &&
      typeof value === "object" &&
      "thread" in value &&
      "messages" in value &&
      "runs" in value
  );
}

function isAuthorConversationResult(value: unknown): value is AuthorConversationResult {
  return Boolean(
    value &&
      typeof value === "object" &&
      "threadId" in value &&
      "runId" in value &&
      "status" in value &&
      "operation" in value
  );
}

function errorMessageFromBody(body: ApiErrorBody | AuthorThreadSnapshot | AuthorConversationResult | null, fallback: string) {
  if (!body || typeof body !== "object") return fallback;
  if ("error" in body && typeof body.error?.message === "string") return body.error.message;
  if ("message" in body && typeof body.message === "string") return body.message;
  return fallback;
}

function redirectToSignIn() {
  window.location.href = `/sign-in?redirect_url=${encodeURIComponent(window.location.href)}`;
}

function threadIdFromAuthorUrl(href: string): string | null {
  try {
    const value = new URL(href).searchParams.get("thread");
    return value?.trim() || null;
  } catch {
    return null;
  }
}

async function writeClipboardText(text: string) {
  if (typeof navigator !== "undefined" && navigator.clipboard?.writeText) {
    try {
      await navigator.clipboard.writeText(text);
      return;
    } catch (error) {
      if (fallbackCopyText(text)) return;
      throw error;
    }
  }
  if (fallbackCopyText(text)) return;
  throw new Error("Clipboard write permission denied.");
}

function fallbackCopyText(text: string) {
  if (typeof document === "undefined") return false;
  const textarea = document.createElement("textarea");
  textarea.value = text;
  textarea.setAttribute("readonly", "");
  textarea.style.position = "fixed";
  textarea.style.left = "-9999px";
  textarea.style.top = "0";
  document.body.appendChild(textarea);
  textarea.select();
  try {
    return document.execCommand("copy");
  } catch {
    return false;
  } finally {
    document.body.removeChild(textarea);
  }
}

function triggerDownload(blob: Blob, filename: string) {
  const url = URL.createObjectURL(blob);
  const anchor = document.createElement("a");
  anchor.href = url;
  anchor.download = filename;
  document.body.appendChild(anchor);
  anchor.click();
  document.body.removeChild(anchor);
  window.setTimeout(() => URL.revokeObjectURL(url), 500);
}

function exportFilenameForFormat(filenameBase: string, targetFormat: Format) {
  const extension =
    targetFormat === "yaml" ? "yaml" : targetFormat === "markdown" ? "md" : "json";
  const base = stripAuthorSourceExtension(filenameBase).trim() || "author-output";
  return `${base}.opf.${extension}`;
}

function exportMimeType(targetFormat: Format) {
  if (targetFormat === "yaml") return "application/yaml";
  if (targetFormat === "markdown") return "text/markdown";
  return "application/json";
}

function exportFormatLabel(targetFormat: Format) {
  if (targetFormat === "yaml") return "YAML";
  if (targetFormat === "markdown") return "Markdown";
  return "JSON";
}
